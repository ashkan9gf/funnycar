    43→/** 'rgba(...)' یا '#hex' → رنگ + آلفا */
    44→function parseColor(c: string): { color: THREE.Color; alpha: number } {
    45→  const m = /rgba?\(([^)]+)\)/.exec(c);
    46→  if (m) {
    47→    const p = m[1].split(',').map((s) => parseFloat(s.trim()));
    48→    return { color: new THREE.Color(p[0] / 255, p[1] / 255, p[2] / 255), alpha: p[3] ?? 1 };
    49→  }
    50→  return { color: new THREE.Color(c), alpha: 1 };
    51→}

   219→  // ============ دنیای ثابت ============
   220→  private buildStaticWorld(): void {
   221→    const S = WORLD_SIZE;
   222→
   223→    // زمین پایه (شن)
   224→    const base = new THREE.Mesh(
   225→      new THREE.PlaneGeometry(S + 2200, S + 2200),
   226→      new THREE.MeshLambertMaterial({ color: '#C9B98A' }),
   227→    );
   228→    base.rotation.x = -Math.PI / 2;
   229→    base.position.set(S / 2, 0, S / 2);
   230→    base.receiveShadow = true;
   231→    this.scene.add(base);
   232→
   233→    // جاده‌ها (۴ عمودی + ۴ افقی)
   234→    const roadTex = getRoadStraightTexture();
   235→    roadTex.repeat.set(1, S / 256);
   236→    const roadMat = new THREE.MeshLambertMaterial({ map: roadTex });
   237→    const roadGeo = new THREE.PlaneGeometry(256, S);
   238→    roadGeo.rotateX(-Math.PI / 2);
   239→    for (let i = 0; i < 4; i++) {
   240→      const cx = i * 1024 + 128;
   241→      const v = new THREE.Mesh(roadGeo, roadMat);
   242→      v.position.set(cx, 1, S / 2);
   243→      v.receiveShadow = true;
   244→      this.scene.add(v);
   245→      const h = new THREE.Mesh(roadGeo, roadMat);
   246→      h.position.set(S / 2, 1, cx);
   247→      h.rotation.y = Math.PI / 2;
   248→      h.receiveShadow = true;
   249→      this.scene.add(h);
   250→    }
   251→
   252→    // تقاطع‌ها (آسفالت ساده روی خط‌کشی)
   253→    const asTex = getAsphaltPlainTexture();
   254→    const asMat = new THREE.MeshLambertMaterial({ map: asTex });
   255→    const crossGeo = new THREE.PlaneGeometry(256, 256);
   256→    crossGeo.rotateX(-Math.PI / 2);
   257→    for (let i = 0; i < 4; i++) {
   258→      for (let j = 0; j < 4; j++) {
   259→        const c = new THREE.Mesh(crossGeo, asMat);
   260→        c.position.set(i * 1024 + 128, 1.6, j * 1024 + 128);
   261→        c.receiveShadow = true;
   262→        this.scene.add(c);
   263→      }
   264→    }
   265→
   266→    // بلوک‌ها: پیاده‌رو + مرکز (چمن پارک / شن ساختمان‌ها)
   267→    const swMat = new THREE.MeshLambertMaterial({ map: getSidewalkTexture() });
   268→    const grassMat = new THREE.MeshLambertMaterial({ map: getGrassTexture() });
   269→    const yardMat = new THREE.MeshLambertMaterial({ color: '#d4c49a' });
   270→    const swGeo = new THREE.PlaneGeometry(768, 768);
   271→    swGeo.rotateX(-Math.PI / 2);
   272→    const innerGeo = new THREE.PlaneGeometry(640, 640);
   273→    innerGeo.rotateX(-Math.PI / 2);
   274→    for (let i = 0; i < 3; i++) {
   275→      for (let j = 0; j < 3; j++) {
   276→        const bx = i * 1024 + 256;
   277→        const by = j * 1024 + 256;
   278→        const sw = new THREE.Mesh(swGeo, swMat);
   279→        sw.position.set(bx + 384, 0.8, by + 384);
   280→        sw.receiveShadow = true;
   281→        this.scene.add(sw);
   282→        const isPark = i === 1 && j === 1;
   283→        const inner = new THREE.Mesh(innerGeo, isPark ? grassMat : yardMat);
   284→        inner.position.set(bx + 384, 1.2, by + 384);
   285→        inner.receiveShadow = true;
   286→        this.scene.add(inner);
   287→      }
   288→    }
   289→
   290→    // ساختمان‌ها و پروپ‌ها از داده World
   291→    for (const p of this.world.props) {
   292→      if (p.kind === 'building') this.buildBuilding(p);
   293→      else this.buildProp(p);
   294→    }
   295→
   296→    // جدول مرزی (دید بصری دیوار محدوده)
   297→    const curbMat = new THREE.MeshLambertMaterial({ color: '#a89a72' });
   298→    const curbLen = S + 60;
   299→    const curbH = 30;
   300→    const walls: Array<[number, number, number, number]> = [
   301→      [S / 2, -20, curbLen, 40],
   302→      [S / 2, S + 20, curbLen, 40],
   303→      [-20, S / 2, 40, curbLen],
   304→      [S + 20, S / 2, 40, curbLen],
   305→    ];
   306→    for (const [wx, wz, ww, wd] of walls) {
   307→      const m = new THREE.Mesh(new THREE.BoxGeometry(ww, curbH, wd), curbMat);
   308→      m.position.set(wx, curbH / 2, wz);
   309→      m.castShadow = true;
   310→      m.receiveShadow = true;
   311→      this.scene.add(m);
   312→    }
   313→  }

   315→  private buildBuilding(p: { x: number; y: number; w: number; h: number; assetId?: string; landmark?: boolean }): void {
   316→    const cx = p.x + p.w / 2;
   317→    const cz = p.y + p.h / 2;
   318→
   319→    if (p.landmark) {
   320→      // پمپ بنزین: مغازه + سقف سایبان + ستون‌ها + پمپ‌ها
   321→      const shopH = RENDER3D.gasStationShopHeight;
   322→      const shopTex = getFacadeTextures('env_building_shop_01');
   323→      const shopMats = shopTex.map((t) => new THREE.MeshLambertMaterial({ map: t }));
   324→      const shop = new THREE.Mesh(new THREE.BoxGeometry(320, shopH, 300), shopMats);
   325→      shop.position.set(cx + p.w * 0.2, shopH / 2, cz - p.h * 0.18);
   326→      shop.castShadow = true; shop.receiveShadow = true;
   327→      this.scene.add(shop);
   328→
   329→      const canopy = new THREE.Mesh(
   330→        new THREE.BoxGeometry(p.w * 0.82, 26, p.h * 0.66),
   331→        new THREE.MeshLambertMaterial({ color: '#e2574c' }),
   332→      );
   333→      canopy.position.set(cx - p.w * 0.08, 178, cz + p.h * 0.16);
   334→      canopy.castShadow = true;
   335→      this.scene.add(canopy);
   336→      const poleGeo = new THREE.CylinderGeometry(12, 12, 168, 8);
   337→      const poleMat = new THREE.MeshLambertMaterial({ color: '#8f8f8a' });
   338→      for (const dx of [-p.w * 0.36, p.w * 0.2]) {
   339→        for (const dz of [-p.h * 0.14, p.h * 0.44]) {
   340→          const pole = new THREE.Mesh(poleGeo, poleMat);
   341→          pole.position.set(cx - p.w * 0.08 + dx, 84, cz + p.h * 0.16 + dz);
   342→          pole.castShadow = true;
   343→          this.scene.add(pole);
   344→        }
   345→      }
   346→      const pumpGeo = new THREE.BoxGeometry(34, 66, 22);
   347→      const pumpMat = new THREE.MeshLambertMaterial({ color: '#d8d3c8' });
   348→      for (const dx of [-p.w * 0.2, p.w * 0.04]) {
   349→        const pump = new THREE.Mesh(pumpGeo, pumpMat);
   350→        pump.position.set(cx - p.w * 0.08 + dx, 33, cz + p.h * 0.16);
   351→        pump.castShadow = true;
   352→        this.scene.add(pump);
   353→      }
   354→      return;
   355→    }
   356→
   357→    const h = RENDER3D.buildingHeights[p.assetId as keyof typeof RENDER3D.buildingHeights] ?? 340;
   358→    const tex = getFacadeTextures(p.assetId ?? 'env_building_home_01');
   359→    const mats = tex.map((t) => new THREE.MeshLambertMaterial({ map: t }));
   360→    const box = new THREE.Mesh(new THREE.BoxGeometry(p.w, h, p.h), mats);
   361→    box.position.set(cx, h / 2, cz);
   362→    box.castShadow = true;
   363→    box.receiveShadow = true;
   364→    this.scene.add(box);
   365→  }

   367→  private buildProp(p: { kind: string; x: number; y: number; w: number; h: number }): void {
   368→    const cx = p.x + p.w / 2;
   369→    const cz = p.y + p.h / 2;
   370→    const r = Math.min(p.w, p.h) / 2;
   371→    switch (p.kind) {
   372→      case 'tree': {
   373→        const trunk = new THREE.Mesh(
   374→          new THREE.CylinderGeometry(r * 0.16, r * 0.22, r * 1.1, 8),
   375→          new THREE.MeshLambertMaterial({ color: '#8a5a33' }),
   376→        );
   377→        trunk.position.set(cx, r * 0.55, cz);
   378→        trunk.castShadow = true;
   379→        const blob = new THREE.Mesh(
   380→          new THREE.IcosahedronGeometry(r * 0.92, 0),
   381→          new THREE.MeshLambertMaterial({ color: '#5FA84B' }),
   382→        );
   383→        blob.position.set(cx, r * 1.5, cz);
   384→        blob.castShadow = true;
   385→        const blob2 = new THREE.Mesh(
   386→          new THREE.IcosahedronGeometry(r * 0.55, 0),
   387→          new THREE.MeshLambertMaterial({ color: '#8AC926' }),
   388→        );
   389→        blob2.position.set(cx + r * 0.3, r * 1.9, cz - r * 0.2);
   390→        blob2.castShadow = true;
   391→        this.scene.add(trunk, blob, blob2);
   392→        break;
   393→      }
   394→      case 'trash': {
   395→        const bin = new THREE.Mesh(
   396→          new THREE.CylinderGeometry(r * 0.8, r * 0.66, r * 1.5, 10),
   397→          new THREE.MeshLambertMaterial({ color: '#4F772D' }),
   398→        );
   399→        bin.position.set(cx, r * 0.75, cz);
   400→        bin.castShadow = true;
   401→        this.scene.add(bin);
   402→        break;
   403→      }
   404→      case 'pole': {
   405→        const pole = new THREE.Mesh(
   406→          new THREE.CylinderGeometry(r * 0.14, r * 0.16, r * 4.2, 8),
   407→          new THREE.MeshLambertMaterial({ color: '#8D99AE' }),

   412→          new THREE.SphereGeometry(r * 0.3, 10, 8),
   413→          new THREE.MeshBasicMaterial({ color: '#EF476F' }),
   414→        );
   415→        lamp.position.set(cx, r * 4.3, cz);
   416→        this.scene.add(pole, lamp);
   417→        break;
   418→      }
   419→      case 'cone': {
   420→        const cone = new THREE.Mesh(
   421→          new THREE.ConeGeometry(r * 0.75, r * 1.7, 10),
   422→          new THREE.MeshLambertMaterial({ color: '#F4A261' }),
   423→        );
   424→        cone.position.set(cx, r * 0.85, cz);
   425→        cone.castShadow = true;
   426→        const band = new THREE.Mesh(
   427→          new THREE.CylinderGeometry(r * 0.42, r * 0.5, r * 0.28, 10),
   428→          new THREE.MeshLambertMaterial({ color: '#FFFFFF' }),
   429→        );
   430→        band.position.set(cx, r * 1.0, cz);
   431→        this.scene.add(cone, band);
   432→        break;
   433→      }
   434→    }
   435→  }

   446→    const head = new THREE.Mesh(new THREE.SphereGeometry(s * 0.3, 10, 8), fur);

   448→    head.castShadow = true;
   449→    const earGeo = new THREE.ConeGeometry(s * 0.1, s * 0.22, 4);
   450→    for (const dx of [-s * 0.16, s * 0.16]) {
   451→      const ear = new THREE.Mesh(earGeo, fur);
   452→      ear.position.set(dx, s * 0.98, -s * 0.42);
   453→      g.add(ear);
   454→    }
   455→    const eyeGeo = new THREE.SphereGeometry(s * 0.06, 6, 6);
   456→    const eyeMat = new THREE.MeshBasicMaterial({ color: '#F2B705' });

   458→      const eye = new THREE.Mesh(eyeGeo, eyeMat);

   463→    tail.position.set(0, s * 0.62, s * 0.52);

   487→        vy: 130 + Math.random() * 170,
   488→        life: 0.5 + Math.random() * 0.3,
   489→        maxLife: 0.8,
   490→        size: 6 + Math.random() * 8,

   505→  /** به‌روزرسانی افکت‌ها (هر فریم، مستقل از گام فیزیک) */
   506→  update(dt: number): void {
   507→    // ذرات
   508→    let anyAlive = false;
   509→    let anyDead = false;
   510→    for (const p of this.particles) {
   511→      if (!p.active) continue;
   512→      p.life -= dt;
   513→      if (p.life <= 0) { p.active = false; anyDead = true; continue; }
   514→      p.x += p.vx * dt;
   515→      p.z += p.vz * dt;
   516→      p.y += p.vy * dt;
   517→      p.vy -= 560 * dt;
   518→      p.vx *= 0.94;
   519→      p.vz *= 0.94;
   520→      if (p.y < 12) { p.y = 12; p.vy *= -0.3; }
   521→      anyAlive = true;
   522→    }
   523→    if (anyDead) this.particles = this.particles.filter((p) => p.active);
   524→    // رد لاستیک
   525→    for (const s of this.skidData) {
   526→      if (s.life > 0) s.life -= dt;
   527→    }
   528→    // حباب‌ها
   529→    for (let i = this.bubbles.length - 1; i >= 0; i--) {
   530→      this.bubbles[i].ttl -= dt;
   531→      if (this.bubbles[i].ttl <= 0) this.bubbles.splice(i, 1);
   532→    }
   533→    if (anyAlive || anyDead) this.syncParticles();
   534→  }

   677→  private updateChaseCamera(car: Vehicle, dt: number, _t: number): void {
   678→    // نرم‌سازی زاویه (کوتاه‌ترین مسیر)
   679→    let diff = car.angle - this.camAngle;
   680→    while (diff > Math.PI) diff -= Math.PI * 2;
   681→    while (diff < -Math.PI) diff += Math.PI * 2;
   682→    this.camAngle += diff * Math.min(1, RENDER3D.camAngleLerp * dt);
   683→
   684→    const fx = Math.sin(this.camAngle);
   685→    const fz = -Math.cos(this.camAngle);
   686→    const back = RENDER3D.camBack + Math.abs(car.speed) * RENDER3D.camBackPerSpeed;
   687→    const desired = this.tmpP.set(car.x - fx * back, RENDER3D.camHeight, car.y - fz * back);
   688→    const k = 1 - Math.exp(-RENDER3D.camPosLerp * dt);
   689→    this.cam3.position.lerp(desired, k);
   690→
   691→    // لرزش از دوربین دوبعدی (برخورد)
   692→    const sh = this.camera.getShake();
   693→    this.cam3.position.x += sh.x * 0.9;
   694→    this.cam3.position.y += Math.abs(sh.y) * 0.5;
   695→    this.cam3.position.z += sh.y * 0.9;
   696→
   697→    // نگاه به جلوی ماشین
   698→    const look = this.tmpV.set(car.x + fx * 260, 60, car.y + fz * 260);
   699→    this.cam3.lookAt(look);
   700→  }

   702→  private syncParticles(): void {
   703→    let n = 0;
   704→    for (const p of this.particles) {
   705→      if (!p.active) continue;
   706→      const k = Math.max(0, p.life / p.maxLife);
   707→      this.tmpQ.identity();
   708→      this.tmpM.compose(this.tmpV.set(p.x, p.y, p.z), this.tmpQ, this.tmpS.setScalar(p.size * (0.5 + k * 0.5)));
   709→      this.particleMesh.setMatrixAt(n, this.tmpM);
   710→      this.particleMesh.setColorAt(n, p.color);
   711→      n++;
   712→      if (n >= RENDER3D.particleMax) break;
   713→    }
   714→    this.particleMesh.count = n;
   715→    this.particleMesh.instanceMatrix.needsUpdate = true;
   716→    if (this.particleMesh.instanceColor) this.particleMesh.instanceColor.needsUpdate = true;
   717→  }

   727→      sy: (-this.projV.y * 0.5 + 0.5) * this.overlayH,
   728→      ok: true,
   729→    };

   811→   * حباب RTL با پرتره — نسخه صفحه‌ای (مختصات پیکسلی اورلی)
   812→   * هم‌ظاهر WorldRenderer.drawBubbleBox دوبعدی

   814→  private drawBubbleBox(
   815→    ctx: CanvasRenderingContext2D,
   816→    text: string,
   817→    anchorX: number,
   818→    anchorY: number,
   819→    anchorH = 100,
   820→    portraitId?: string,
   821→  ): void {
   822→    ctx.save();
   823→    ctx.direction = 'rtl';
   824→    ctx.font = '700 24px Vazirmatn, Tahoma, sans-serif';
   825→    ctx.textBaseline = 'middle';
   826→    ctx.textAlign = 'center';
   827→
   828→    const maxTextW = 260;
   829→    const words = text.split(' ');
   830→    const lines: string[] = [];
   831→    let current = '';
   832→    for (const w of words) {
   833→      const test = current ? `${current} ${w}` : w;
   834→      if (ctx.measureText(test).width > maxTextW && current) {
   835→        lines.push(current);
   836→        current = w;
   837→      } else current = test;
   838→    }
   839→    if (current) lines.push(current);
   840→
   841→    const lineH = 32;
   842→    const portraitR = 26;
   843→    const bubbleW = Math.min(maxTextW + portraitR * 2 + 40, 360);
   844→    const bubbleH = Math.max(72, lines.length * lineH + 34);
   845→
   846→    let bx = anchorX - bubbleW / 2;
   847→    let by = anchorY - anchorH / 2 - bubbleH - 22;
   848→    let tailBelow = false;
   849→    if (by < 8) {
   850→      by = anchorY + anchorH / 2 + 22;
   851→      tailBelow = true;
   852→    }
   853→    if (by + bubbleH > this.overlayH - 8) by = this.overlayH - 8 - bubbleH;
   854→    if (by < 8) by = 8;
   855→    bx = Math.min(Math.max(bx, 8), this.overlayW - 8 - bubbleW);
   856→    if (bx < 8) bx = 8;
   857→
   858→    // دم
   859→    ctx.fillStyle = '#FFF9EC';
   860→    ctx.strokeStyle = '#3A2E2A';
   861→    ctx.lineWidth = 4;
   862→    ctx.beginPath();
   863→    if (tailBelow) {
   864→      ctx.moveTo(anchorX - 13, by + 2);
   865→      ctx.lineTo(anchorX, by - 20);
   866→      ctx.lineTo(anchorX + 13, by + 2);
   867→    } else {
   868→      ctx.moveTo(anchorX - 13, by + bubbleH - 2);
   869→      ctx.lineTo(anchorX, by + bubbleH + 20);
   870→      ctx.lineTo(anchorX + 13, by + bubbleH - 2);
   871→    }
   872→    ctx.closePath();
   873→    ctx.fill();
   874→    ctx.stroke();
   875→
   876→    // بدنه
   877→    ctx.beginPath();
   878→    const r = 18;
   879→    ctx.moveTo(bx + r, by);
   880→    ctx.arcTo(bx + bubbleW, by, bx + bubbleW, by + bubbleH, r);
   881→    ctx.arcTo(bx + bubbleW, by + bubbleH, bx, by + bubbleH, r);
   882→    ctx.arcTo(bx, by + bubbleH, bx, by, r);
   883→    ctx.arcTo(bx, by, bx + bubbleW, by, r);
   884→    ctx.closePath();
   885→    ctx.fill();
   886→    ctx.stroke();
   887→
   888→    // پرتره دایره‌ای
   889→    const pid = portraitId ?? 'char_driver_01_idle';
   890→    const pcx = bx + bubbleW - portraitR - 14;
   891→    const pcy = by + bubbleH / 2;
   892→    const asset = this.assets.get(pid);
   893→    const img = asset instanceof AudioBuffer ? undefined : asset;
   894→    if (img) {
   895→      ctx.save();
   896→      ctx.beginPath();
   897→      ctx.arc(pcx, pcy, portraitR, 0, Math.PI * 2);
   898→      ctx.clip();
   899→      ctx.drawImage(img, pcx - portraitR, pcy - portraitR, portraitR * 2, portraitR * 2);
   900→      ctx.restore();
   901→    } else {
   902→      ctx.fillStyle = '#E8C39E';
   903→      ctx.beginPath();
   904→      ctx.arc(pcx, pcy, portraitR, 0, Math.PI * 2);
   905→      ctx.fill();
   906→    }
   907→    ctx.strokeStyle = '#3A2E2A';
   908→    ctx.lineWidth = 3.5;
   909→    ctx.beginPath();
   910→    ctx.arc(pcx, pcy, portraitR, 0, Math.PI * 2);
   911→    ctx.stroke();
   912→
   913→    // متن
   914→    ctx.fillStyle = '#3A2E2A';
   915→    const textAreaCenter = bx + (bubbleW - portraitR * 2 - 18) / 2;
   916→    lines.forEach((ln, i) => {
   917→      ctx.fillText(ln, textAreaCenter, by + bubbleH / 2 + (i - (lines.length - 1) / 2) * lineH);
   918→    });
   919→    ctx.restore();
   920→  }

   922→  // ============ اندازه و پاک‌سازی ============

   924→  /** w/h پیکسل فیزیکی کانواس WebGL + dpr برای اورلی */
   925→  resize(pixelW: number, pixelH: number, dpr: number): void {
   926→    this.dpr = dpr;
   927→    this.r3.setPixelRatio(1);
   928→    this.r3.setSize(pixelW, pixelH, false);
   929→    this.overlay.width = pixelW;
   930→    this.overlay.height = pixelH;
   931→    // اندازه CSS اورلی = ۱۰۰٪ والد (در DOM تنظیم شده)
   932→    // ابعاد منطقی برای پروجکشن = پیکسل CSS
   933→    const rect = this.overlay.getBoundingClientRect();
   934→    this.overlayW = rect.width || pixelW / dpr;
   935→    this.overlayH = rect.height || pixelH / dpr;
   936→    // FOV تطبیقی: در صفحه‌های باریک (موبایل عمودی) زاویه دید بازتر می‌شود
   937→    const aspect = this.overlayW / Math.max(1, this.overlayH);
   938→    const extraFov = Math.min(26, Math.max(0, (1 / Math.max(aspect, 0.2) - 1) * 22));
   939→    this.cam3.fov = RENDER3D.fov + extraFov;
   940→    this.cam3.aspect = aspect;
   941→    this.cam3.updateProjectionMatrix();
   942→  }
   943→
   944→  dispose(): void {
   945→    this.scene.traverse((obj) => {
   946→      const mesh = obj as THREE.Mesh;
   947→      if (mesh.geometry) mesh.geometry.dispose();
   948→      const mat = (mesh as unknown as { material?: THREE.Material | THREE.Material[] }).material;
   949→      if (Array.isArray(mat)) mat.forEach((m) => m.dispose());
   950→      else if (mat) mat.dispose();
   951→    });
   952→    // توجه: کانتکت WebGL را نمی‌بندیم (loseContext) — چون کانواس در
   953→    // StrictMode/HMR دوباره استفاده می‌شود و کانتکت همان است.
   954→    this.r3.dispose();
   955→    this.overlay.parentElement?.removeChild(this.overlay);
   956→  }
