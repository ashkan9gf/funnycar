    65→/** پنل تنظیمات صدا (در منو و Pause استفاده می‌شود) */
    66→function SettingsPanel({ game, onClose }: { game: GameInstance; onClose?: () => void }) {
    67→  const [s, setS] = useState<Settings>(game.getSettings());
    68→  const rows: Array<{ key: keyof Settings; label: string; icon: React.ReactNode }> = [
    69→    { key: "music", label: "موسیقی", icon: <Music className="h-5 w-5" /> },
    70→    { key: "sfx", label: "افکت‌ها", icon: <Volume2 className="h-5 w-5" /> },
    71→    { key: "voice", label: "جملات طنز", icon: <Mic className="h-5 w-5" /> },
    72→  ];
    73→  return (
    74→    <div className="w-full space-y-3">
    75→      {rows.map((r) => (
    76→        <div key={r.key} className="flex items-center gap-3">
    77→          <span className="text-[#3a2e2a]">{r.icon}</span>
    78→          <span className="w-24 text-sm font-bold text-[#3a2e2a]">{r.label}</span>
    79→          <input
    80→            type="range"
    81→            className="fc-range flex-1"
    82→            min={0}
    83→            max={1}
    84→            step={0.05}
    85→            value={s[r.key]}
    86→            aria-label={`صدای ${r.label}`}
    87→            onChange={(e) => {
    88→              const v = Number(e.target.value);
    89→              setS((p) => ({ ...p, [r.key]: v }));
    90→              game.setSetting(r.key, v);
    91→            }}
    92→          />
    93→          <span className="w-10 text-center text-sm font-bold text-[#3a2e2a]">{fa(Math.round(s[r.key] * 100))}٪</span>
    94→        </div>
    95→      ))}
    96→      <div className="flex items-center gap-3 opacity-50">
    97→        <span className="text-sm font-bold text-[#3a2e2a]">لرزش گوشی</span>
    98→        <span className="text-xs text-[#3a2e2a]">(فازهای بعدی)</span>
    99→      </div>
   100→      {onClose && (
   101→        <button
   102→          type="button"
   103→          className="fc-btn mt-2 w-full rounded-2xl bg-[#3a2e2a] py-2.5 text-white"
   104→          onPointerDown={(e) => {
   105→            e.preventDefault();
   106→            onClose();
   107→          }}
   108→        >
   109→          بستن
   110→        </button>
   111→      )}
   112→    </div>
   113→  );
   114→}

   143→  const gameRef = useRef<GameInstance | null>(null);
   144→  const steerRef = useRef({ left: false, right: false });

   146→
   147→  /** نمونه بازی در state برای رندر امن React (gameRef فقط در event handlerها استفاده می‌شود) */
   148→  const [game, setGame] = useState<GameInstance | null>(null);
   149→
   150→  const [phase, setPhase] = useState<GameState | "boot">("boot");
   151→  const [bootProgress, setBootProgress] = useState(0);
   152→  const [hud, setHud] = useState<HudData>({ speedKmh: 0, coins: 0, reputation: 50, fps: 60, showFps: false, placeholders: 0, mission: null, health: 100, engineDead: false });
   153→  const [toast, setToast] = useState<string | null>(null);
   154→  const [showSettings, setShowSettings] = useState(false);

   159→  const [missionResult, setMissionResult] = useState<MissionResultData | null>(null);
   160→  /** فاز ۶: گاراژ عمو نقی */
   161→  const [showGarage, setShowGarage] = useState(false);
   162→  const [garage, setGarage] = useState<GarageView | null>(null);

   168→    setShowGarage(true);

   171→  const showToast = useCallback((msg: string) => {
   172→    setToast(msg);
   173→    if (toastTimer.current) clearTimeout(toastTimer.current);
   174→    toastTimer.current = setTimeout(() => setToast(null), 1800);
   175→  }, []);

   178→  const purchase = useCallback(
   179→    (fn: () => { ok: boolean; msg: string }) => {
   180→      const r = fn();
   181→      showToast(r.msg);
   182→      const g = gameRef.current?.listGarage();
   183→      if (g) setGarage(g);
   184→    },

   186→  );
   187→
   188→  // ---------- راه‌اندازی موتور ----------
   189→  useEffect(() => {
   190→    let cancelled = false;
   191→    const canvas = canvasRef.current;
   192→    if (!canvas) return;

   194→      canvas,
   195→      {
   196→        onStateChange: (s) => setPhase(s),
   197→        onHud: (h) => setHud(h),

   224→      applySteer();

   226→    [applySteer],
   227→  );

   229→    (down: boolean) => {
   230→      steerRef.current.right = down;
   231→      applySteer();

   244→
   245→      {/* ============ صفحه بوت ============ */}

   249→            <Loader2 className="h-10 w-10 animate-spin text-[#2A9D8F]" />
   250→            <p className="text-xl font-extrabold text-[#3a2e2a]">در حال باز کردن گاراژ...</p>
   251→            <div className="h-4 w-56 overflow-hidden rounded-full border-4 border-[#3a2e2a] bg-[#FFF3D6]">
   252→              <div className="h-full bg-[#8AC926] transition-all" style={{ width: `${bootProgress}%` }} />
   253→            </div>
   254→            <p className="text-sm font-bold text-[#3a2e2a]">٪{fa(bootProgress)}</p>
   255→          </div>
   256→        </div>
   257→      )}
   258→
   259→      {/* ============ منوی اصلی ============ */}

   266→                <Car className="h-11 w-11 text-[#3a2e2a]" strokeWidth={2.4} />
   267→              </div>
   268→              <h1 className="mt-2 text-5xl font-extrabold text-white drop-shadow-[0_4px_0_rgba(58,46,42,0.9)]">فانی‌کار</h1>
   269→              <p className="text-sm font-bold text-white/90 drop-shadow-[0_2px_0_rgba(58,46,42,0.7)]">
   270→                آبرو می‌بری یا می‌ریزی؟
   271→              </p>
   272→            </div>
   273→
   274→            {/* آمار بازیکن */}
   275→            <div className="flex items-center gap-3">

   286→            {/* دکمه اصلی */}
   287→            <button
   288→              type="button"
   289→              className="fc-btn w-full max-w-xs rounded-3xl bg-[#8AC926] py-4 text-2xl font-extrabold text-[#3a2e2a]"
   290→              onPointerDown={(e) => {
   291→                e.preventDefault();

   355→                  setShowSettings(true);
   356→                }}
   357→              >
   358→                <Settings className="h-5 w-5" />
   359→                تنظیمات

   363→            {/* یادداشت Placeholder (شفافیت — قانون بخش ۳۲ سند مادر) */}
