   306→  /** پایان مأموریت — جایزه/جریمه + ذخیره + رویداد به پوسته */
   307→  private finishMission(
   308→    success: boolean,
   309→    missionId: string,
   310→    baseCoins: number,
   311→    baseRep: number,
   312→    reason: string | null,
   313→    _stats: unknown,
   314→  ): void {
   315→    if (this.state !== 'playing') return;
   316→    const def = this.mission.getDef();
   317→    const title = def?.title ?? missionId;
   318→
   319→    let coins = 0;
   320→    let rep = 0;
   321→    if (success) {
   322→      // بونوس آبروی بالا (سند بخش ۷: ×1.25 وقتی آبرو ≥ ۸۰)
   323→      const repNow = this.save.get().reputation;
   324→      coins = Math.round(baseCoins * (repNow >= MISSIONS.bonusRepThreshold ? MISSIONS.bonusFactor : 1));
   325→      rep = baseRep;
   326→      this.save.mutate((d) => {
   327→        d.coins += coins;
   328→        d.reputation = Math.min(REPUTATION.max, d.reputation + rep);
   329→        if (!d.completedMissions.includes(missionId)) d.completedMissions.push(missionId);
   330→      });
   331→    }
   332→
   333→    this.audio.coin();
   334→    if (success) this.fx.spawnBurst(this.vehicle.x, this.vehicle.y - 30, '#F2B705', 16, 150);
   335→    this.voice.trigger(success ? 'mission_success' : 'mission_fail', performance.now());
   336→
   337→    const line = this.pickLine(success ? 'mission_success' : 'mission_fail');
   338→    this.callbacks.onMissionEnd?.({ success, missionId, title, coins, rep, reason, line });
   339→  }

   372→  /** برخورد بازیکن با خودروی ترافیک — بازیکن کنار می‌رود، ترافیک مکث می‌کند */
   373→  private resolveTrafficCollisions(dt: number): void {
   374→    const v = this.vehicle;
   375→    this.crashCooldown = Math.max(0, this.crashCooldown - dt);
   376→    for (const c of this.traffic.cars) {
   377→      const hw = v.hitW / 2 + c.hitW / 2;
   378→      const hh = v.hitH / 2 + c.hitH / 2;
   379→      const dx = v.x - c.x;
   380→      const dy = v.y - c.y;
   381→      const penX = hw - Math.abs(dx);
   382→      const penY = hh - Math.abs(dy);
   383→      if (penX <= 0 || penY <= 0) continue;
   384→
   385→      const impact = Math.hypot(v.vx, v.vy);
   386→      if (penX < penY) v.x += dx >= 0 ? penX : -penX;
   387→      else v.y += dy >= 0 ? penY : -penY;
   388→      v.speed *= 0.5;
   389→      v.latVel *= 0.4;
   390→      c.stun(TRAFFIC.stunnedSec);
   391→
   392→      if (this.crashCooldown <= 0 && impact > COLLISION.softCrashSpeed) {
   393→        this.crashCooldown = 0.8;
   394→        // برخورد با ترافیک هم برای مأموریت شمرده می‌شود (فاز 4)
   395→        this.mission.onCrash(impact > COLLISION.hardCrashSpeed ? 'hard' : 'soft');
   396→        this.audio.crash(false);
   397→        this.camera.addShake(CAM.maxShake * 0.5);
   398→        this.fx.spawnBurst(v.x, v.y, '#B9B9B9', 10);
   399→        this.save.mutate((d) => {
   400→          d.reputation = Math.max(REPUTATION.min, d.reputation - REPUTATION.trafficCrashPenalty);
   401→        });
   402→        this.voice.trigger('car_crash', performance.now());
   403→        if (Math.random() < 0.5) {
   404→          this.fx.pushBubble(this.pickLine('traffic_honk'), c.x, c.y, 'char_driver_01_idle', 2.4);
   405→        }
   406→      }
   407→      break; // یک برخورد در هر گام کافی است
   408→    }
   409→  }

   411→  // ============ حلقه ============
   412→  private loop = (now: number): void => {
   413→    // اگر نمونه دیگری جای این را گرفته، خودکشی کن (ضد زامبی)
   414→    if (this.disposed || GameManager.live !== this) return;
   415→    this.raf = requestAnimationFrame(this.loop);
   416→
   417→    try {
   418→      const rawDt = Math.min(0.1, (now - this.lastFrame) / 1000 || 0);
   419→      this.lastFrame = now;
   420→
   421→      // FPS
   422→      this.fpsAccum += rawDt;
   423→      this.fpsCount++;
   424→      if (this.fpsAccum >= 0.5) {
   425→        this.fpsValue = Math.round(this.fpsCount / this.fpsAccum);
   426→        this.fpsAccum = 0;
   427→        this.fpsCount = 0;
   428→      }
   429→
   430→      if (this.state === 'playing') {
   431→        this.accumulator += rawDt;
   432→        let steps = 0;
   433→        while (this.accumulator >= FIXED_DT && steps < 5) {
   434→          this.step(FIXED_DT);
   435→          this.accumulator -= FIXED_DT;
   436→          steps++;
   437→        }
   438→        if (steps >= 5) this.accumulator = 0; // جلوگیری از Spiral of Death
   439→        this.voice.update(rawDt);
   440→      }
   441→
   442→      this.fx.update(rawDt);
   443→      this.draw();
   444→      this.emitHud(rawDt);
   445→    } catch (err) {
   446→      // خطای یک فریم هرگز بازی را نمی‌بندد
   447→      console.warn('[Game] frame error:', err);
   448→      this.accumulator = 0;
   449→    }
   450→  };

   452→  private step(dt: number): void {
   453→    const input = this.input.collect();
   454→
   455→    // بوق از کیبورد (H)
   456→    if (input.hornPressed) this.pressHorn();
   457→
   458→    const result = this.vehicle.update(dt, input, this.world.colliders);
   459→    const nowMs = performance.now();
   460→
   461→    // ---------- برخورد با موانع ثابت ----------
   462→    this.crashCooldown = Math.max(0, this.crashCooldown - dt);
   463→    if (result.crashed && this.crashCooldown <= 0) {
   464→      const hard = result.impactSpeed > 260;
   465→      // گزارش برخورد به مأموریت فقط یک‌بار در هر رویداد (نه هر فریم)
   466→      this.mission.onCrash(hard ? 'hard' : 'soft');
   467→      this.crashCooldown = 0.8;
   468→      this.audio.crash(hard);
   469→      this.camera.addShake(hard ? CAM.maxShake : CAM.maxShake * 0.45);
   470→      this.fx.spawnBurst(this.vehicle.x, this.vehicle.y, hard ? '#9B9B9B' : '#D6D6D6', hard ? 14 : 8);
   471→      this.save.mutate((d) => {
   472→        d.reputation = Math.max(REPUTATION.min, d.reputation - (hard ? REPUTATION.hardCrashPenalty : REPUTATION.softCrashPenalty));
   473→      });
   474→      this.voice.trigger('car_crash', nowMs);
   475→    }
   476→
   477→    // ---------- ترمز ----------
   478→    if (input.brake && Math.abs(this.vehicle.speed) > 160) {
   479→      if (!this.brakeSoundLatch) {
   480→        this.audio.brake();
   481→        this.brakeSoundLatch = true;
   482→        this.voice.trigger('hard_brake', nowMs);
   483→      }
   484→    } else if (!input.brake) {
   485→      this.brakeSoundLatch = false;
   486→    }
   487→
   488→    // ---------- درفت ملایم (فاز 2): رد لاستیک + صدای سایش + دود ----------
   489→    if (result.skidding) {
   490→      // نقطه محور عقب (کمی پشت مرکز ماشین)
   491→      const back = this.vehicle.drawH * 0.32;
   492→      const ax = this.vehicle.x - Math.sin(this.vehicle.angle) * back;
   493→      const ay = this.vehicle.y + Math.cos(this.vehicle.angle) * back;
   494→      this.fx.addSkidMark(ax, ay, this.vehicle.angle);
   495→      this.skidSoundTimer -= dt;
   496→      if (this.skidSoundTimer <= 0) {
   497→        this.skidSoundTimer = DRIFT.skidSoundInterval;
   498→        const intensity = Math.min(1, Math.abs(this.vehicle.latVel) / DRIFT.maxLateral + 0.35);
   499→        this.audio.skid(intensity);
   500→        this.fx.spawnBurst(ax, ay, 'rgba(160,160,160,0.9)', 2, 60);
   501→      }
   502→      this.voice.trigger('drift', nowMs);
   503→    } else {
   504→      this.skidSoundTimer = 0;
   505→    }
   506→
   507→    // ---------- سرعت‌گیرها (فاز GTA-IR): «تق» + کندی + لرزش + غبار ----------
   508→    this.bumpCooldown = Math.max(0, this.bumpCooldown - dt);
   509→    if (this.bumpCooldown <= 0 && Math.abs(this.vehicle.speed) > RENDER3D.bumpMinSpeed) {
   510→      for (const sb of this.world.speedBumps) {
   511→        const dx = this.vehicle.x - sb.x;
   512→        const dy = this.vehicle.y - sb.y;
   513→        // باند عبور عرض جاده: جاده افقی = عبور در راستای X
   514→        const inside = sb.horiz
   515→          ? Math.abs(dx) < 24 && Math.abs(dy) < 100
   516→          : Math.abs(dy) < 24 && Math.abs(dx) < 100;
   517→        if (inside) {
   518→          this.bumpCooldown = RENDER3D.bumpCooldownSec;
   519→          this.vehicle.speed *= RENDER3D.bumpSlowFactor;
   520→          this.camera.addShake(RENDER3D.bumpShake);
   521→          this.audio.bump();
   522→          this.fx.spawnBurst(this.vehicle.x, this.vehicle.y, 'rgba(190,170,130,0.9)', 4, 90);
   523→          break;
   524→        }
   525→      }
   526→    }
   527→
   528→    // ---------- ترافیک (فاز 2) ----------
   529→    this.traffic.update(dt, this.vehicle);
   530→    this.resolveTrafficCollisions(dt);
   531→
   532→    // ---------- رویدادهای تصادفی (فاز 2) ----------
   533→    this.randomEvents.update(dt, this.vehicle);
   534→
   535→    // ---------- مأموریت (فاز 4) ----------
   536→    this.mission.update(dt, this.vehicle);
   537→    // دود ماشین دودکن (m02) — پف دود تا برسد تعمیرگاه
   538→    if (this.mission.getState() === 'active' && this.mission.getDef()?.id === 'm02_smoking_car') {
   539→      this.smokeTimer -= dt;
   540→      if (this.smokeTimer <= 0) {
   541→        this.smokeTimer = MISSIONS.smokeIntervalSec;
   542→        const back = this.vehicle.drawH * 0.42;
   543→        const sx = this.vehicle.x - Math.sin(this.vehicle.angle) * back;
   544→        const sy = this.vehicle.y + Math.cos(this.vehicle.angle) * back;
   545→        this.fx.spawnBurst(sx, sy, 'rgba(120,120,125,0.75)', 1, 40);
   546→      }
   547→    }
   548→
   549→    // ---------- سکه ----------
   550→    const got = this.world.collectCoins(this.vehicle.x, this.vehicle.y, 60);
   551→    if (got > 0) {
   552→      this.audio.coin();
   553→      this.fx.spawnBurst(this.vehicle.x, this.vehicle.y - 30, '#F2B705', 10, 120);
   554→      this.save.mutate((d) => { d.coins += got; });
   555→      this.voice.trigger('coin_pickup', nowMs);
   556→    }
   557→
   558→    // ---------- موتور ----------
   559→    this.audio.setEngineSpeed(Math.abs(this.vehicle.speed) / VEHICLE_PHYSICS.maxSpeed);
   560→
   561→    // ---------- رویداد پس‌زمینه ----------
   562→    this.idleTimerSec -= dt;
   563→    if (this.idleTimerSec <= 0) {
   564→      this.idleTimerSec = IDLE_EVENTS.minIntervalSec + Math.random() * (IDLE_EVENTS.maxIntervalSec - IDLE_EVENTS.minIntervalSec);
   565→      this.voice.trigger('idle_funny', nowMs);
   566→    }
   567→
   568→    // ---------- دوربین ----------
   569→    if (this.ctx) {
   570→      this.camera.halfW = RENDER.logicalWidth / 2;
   571→      this.camera.halfH = this.ctx.canvas.height / (this.scale * 2);
   572→    } else {
   573→      // حالت سه‌بعدی: نیم‌اندازه‌های منطقی صرفاً برای clamp حباب‌ها
   574→      this.camera.halfW = RENDER.logicalWidth / 2;
   575→      this.camera.halfH = 640;
   576→    }
   577→    this.camera.update(dt, this.vehicle.x, this.vehicle.y, this.vehicle.vx, this.vehicle.vy);
   578→  }
   579→
   580→  // ============ رسم ============
   581→  private draw(): void {
   582→    // ---------- مسیر سه‌بعدی (فاز 3D) ----------
   583→    if (this.renderer3d) {
   584→      this.draw3DFrame();
   585→      return;
   586→    }
   587→    // ---------- مسیر دوبعدی (fallback) ----------
   588→    const ctx = this.ctx;
   589→    if (!ctx || !this.renderer) return;
   590→    const cw = ctx.canvas.width;
   591→    const ch = ctx.canvas.height;
   592→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   593→    ctx.fillStyle = '#C9B98A';
   594→    ctx.fillRect(0, 0, cw, ch);
   595→
   596→    // scale منطقی: پیکسل‌های فیزیکی بر واحد منطقی
   597→    this.scale = cw / RENDER.logicalWidth;
   598→
   599→    ctx.scale(this.scale, this.scale);
   600→    this.camera.halfW = RENDER.logicalWidth / 2;
   601→    this.camera.halfH = ch / this.scale / 2;
   602→
   603→    this.renderer.draw(
   604→      ctx,
   605→      this.vehicle,
   606→      this.traffic.cars,
   607→      this.randomEvents.cat,
   608→      this.randomEvents.cones,
   609→      this.mission.getPotholes(),
   610→      this.mission.getState() === 'active' ? this.mission.getDef()?.target ?? null : null,
   611→    );
   612→    // حباب گفتار با مختصات دنیاست — باید با همان ترنسلیت دوربین رسم شود
   613→    // (renderer.draw ترنسلیت را داخل save/restore خودش دارد و بعد از restore حذف می‌شود)
   614→    ctx.save();
   615→    ctx.translate(-this.camera.offsetX(), -this.camera.offsetY());
   616→    this.drawSpeechBubble(ctx);
   617→    ctx.restore();
   618→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   619→
   620→    // فلش جهت‌نمای هدف وقتی بیرون کادر است (فضای صفحه — فاز 4)
   621→    if (this.mission.getState() === 'active') {
   622→      const def = this.mission.getDef();
   623→      if (def) this.renderer.drawOffscreenArrow(ctx, def.target, this.vehicle, cw, ch, this.scale);
   624→    }
   625→  }

   627→  /** یک فریم کامل سه‌بعدی: صحنه + اورلی (حباب‌ها و فلش هدف) */
   628→  private draw3DFrame(): void {
   629→    const active = this.mission.getState() === 'active';
   630→    const def = active ? this.mission.getDef() : null;
   631→    const speech = this.voice.current;
   632→    this.renderer3d!.draw3D({
   633→      car: this.vehicle,
   634→      traffic: this.traffic.cars,
   635→      cat: this.randomEvents.cat,
   636→      cones: this.randomEvents.cones,
   637→      potholes: this.mission.getPotholes(),
   638→      target: def?.target ?? null,
   639→      playerBubble: speech
   640→        ? {
   641→            line: speech.line,
   642→            x: this.vehicle.x,
   643→            y: this.vehicle.y,
   644→            portrait: SPEAKER_PORTRAIT[speech.speaker] ?? 'char_driver_01_idle',
   645→          }
   646→        : null,
   647→    });
   648→  }
   649→
   650→  /** حباب گفتار فارسی (RTL) بازیکن — فقط مسیر دوبعدی (در سه‌بعدی اورلی جای آن را می‌گیرد) */
   651→  private drawSpeechBubble(ctx: CanvasRenderingContext2D): void {
   652→    const info = this.voice.current;
   653→    if (!info || !this.renderer) return;
   654→    const car = this.vehicle;
   655→    // فاز 2: همه گوینده‌ها فعلاً پرتره راننده را دارند؛ با رسیدن پرتره‌های
   656→    // Batch 2-4 کافی است SPEAKER_PORTRAIT در manifest به‌روز شود
   657→    this.renderer.drawBubbleBox(ctx, info.line, car.x, car.y, car.drawH, 'char_driver_01_idle');
   658→  }

   660→  // ============ HUD ============
   661→  private emitHud(dt: number): void {
   662→    this.hudThrottle += dt;
   663→    if (this.hudThrottle < 0.12) return;
   664→    this.hudThrottle = 0;
   665→    this.callbacks.onHud?.({
   666→      speedKmh: Math.round(Math.abs(this.vehicle.speed) / 3),
   667→      coins: this.save.get().coins,
   668→      reputation: this.save.get().reputation,
   669→      fps: this.fpsValue,
   670→      showFps: this.showFps,
   671→      placeholders: this.assets.getReport().placeholders.length,
   672→      mission: this.mission.hud(this.vehicle),
   673→    });
   674→  }

   676→  // ============ رویدادهای صفحه ============
   677→  private handleResize = (): void => {
   678→    const dpr = Math.min(RENDER.maxDevicePixelRatio, window.devicePixelRatio || 1);
   679→    const w = this.canvas.clientWidth || window.innerWidth;
   680→    const h = this.canvas.clientHeight || window.innerHeight;
   681→    this.canvas.width = Math.round(w * dpr);
   682→    this.canvas.height = Math.round(h * dpr);
   683→    // رندر سه‌بعدی هم باید بداند (کانواس WebGL + اورلی + aspect دوربین)
   684→    if (this.renderer3d) this.renderer3d.resize(this.canvas.width, this.canvas.height, dpr);
   685→  };
   686→
   687→  private onKeyDown = (e: KeyboardEvent): void => {
   688→    const k = e.key.toLowerCase();
   689→    if (k === 'p' || k === 'escape') {
   690→      if (this.state === 'playing') this.pause();
   691→      else if (this.state === 'paused') this.resume();
   692→    } else if (k === 'f') {
   693→      this.toggleDebugFps();
   694→    }
   695→  };
   696→
   697→  private onVisibility = (): void => {
   698→    if (document.hidden && this.state === 'playing') this.pause();
   699→  };
   700→
   701→  private onBeforeUnload = (): void => {
   702→    this.save.flush();
   703→  };
   704→
   705→  dispose(): void {
   706→    this.disposed = true;
   707→    if (GameManager.live === this) GameManager.live = null;
   708→    cancelAnimationFrame(this.raf);
   709→    this.audio.stopEngine();
   710→    this.input.detach();
   711→    this.renderer3d?.dispose(); // حذف اورلی + آزادسازی کانتکست GPU
   712→    window.removeEventListener('resize', this.handleResize);
   713→    window.removeEventListener('keydown', this.onKeyDown);
   714→    document.removeEventListener('visibilitychange', this.onVisibility);
   715→    window.removeEventListener('beforeunload', this.onBeforeUnload);
   716→    this.save.flush();
   717→  }
