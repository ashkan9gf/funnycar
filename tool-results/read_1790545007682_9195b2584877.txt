    48→/** 'rgba(...)' یا '#hex' → رنگ + آلفا */
    49→function parseColor(c: string): { color: THREE.Color; alpha: number } {
    50→  const m = /rgba?\(([^)]+)\)/.exec(c);
    51→  if (m) {
    52→    const p = m[1].split(',').map((s) => parseFloat(s.trim()));
    53→    return { color: new THREE.Color(p[0] / 255, p[1] / 255, p[2] / 255), alpha: p[3] ?? 1 };
    54→  }
    55→  return { color: new THREE.Color(c), alpha: 1 };
    56→}

   331→  }

   345→  private buildBuilding(p: { x: number; y: number; w: number; h: number; assetId?: string; landmark?: boolean }): void {
   346→    const cx = p.x + p.w / 2;
   347→    const cz = p.y + p.h / 2;
   348→
   349→    if (p.landmark) {
   350→      // پمپ بنزین: مغازه + سقف سایبان + ستون‌ها + پمپ‌ها
   351→      const shopH = RENDER3D.gasStationShopHeight;
   352→      const shopTex = getFacadeTextures('env_building_shop_01');
   353→      const shopMats = shopTex.map((t) => new THREE.MeshLambertMaterial({ map: t }));
   354→      const shop = new THREE.Mesh(new THREE.BoxGeometry(320, shopH, 300), shopMats);
   355→      shop.position.set(cx + p.w * 0.2, shopH / 2, cz - p.h * 0.18);
   356→      shop.castShadow = true; shop.receiveShadow = true;
   357→      this.scene.add(shop);
   358→
   359→      const canopy = new THREE.Mesh(
   360→        new THREE.BoxGeometry(p.w * 0.82, 26, p.h * 0.66),
   361→        new THREE.MeshLambertMaterial({ color: '#e2574c' }),
   362→      );
   363→      canopy.position.set(cx - p.w * 0.08, 178, cz + p.h * 0.16);
   364→      canopy.castShadow = true;
   365→      this.scene.add(canopy);
   366→      const poleGeo = new THREE.CylinderGeometry(12, 12, 168, 8);
   367→      const poleMat = new THREE.MeshLambertMaterial({ color: '#8f8f8a' });
   368→      for (const dx of [-p.w * 0.36, p.w * 0.2]) {
   369→        for (const dz of [-p.h * 0.14, p.h * 0.44]) {
   370→          const pole = new THREE.Mesh(poleGeo, poleMat);
   371→          pole.position.set(cx - p.w * 0.08 + dx, 84, cz + p.h * 0.16 + dz);
   372→          pole.castShadow = true;
   373→          this.scene.add(pole);
   374→        }
   375→      }
   376→      const pumpGeo = new THREE.BoxGeometry(34, 66, 22);
   377→      const pumpMat = new THREE.MeshLambertMaterial({ color: '#d8d3c8' });
   378→      for (const dx of [-p.w * 0.2, p.w * 0.04]) {
   379→        const pump = new THREE.Mesh(pumpGeo, pumpMat);
   380→        pump.position.set(cx - p.w * 0.08 + dx, 33, cz + p.h * 0.16);
   381→        pump.castShadow = true;
   382→        this.scene.add(pump);
   383→      }
   384→      return;
   385→    }
   386→
   387→    const h = RENDER3D.buildingHeights[p.assetId as keyof typeof RENDER3D.buildingHeights] ?? 340;
   388→    const tex = getFacadeTextures(p.assetId ?? 'env_building_home_01');
   389→    const mats = tex.map((t) => new THREE.MeshLambertMaterial({ map: t }));
   390→    const box = new THREE.Mesh(new THREE.BoxGeometry(p.w, h, p.h), mats);
   391→    box.position.set(cx, h / 2, cz);
   392→    box.castShadow = true;
   393→    box.receiveShadow = true;
   394→    this.scene.add(box);
   395→    // خوشه پشت‌بام ایرانی: ظرف ماهواره + منبع آب + آنتن چتری + کولر آبی
   396→    const roof = buildRooftopCluster(p.w, p.h, Math.round(cx * 7 + cz * 13));
   397→    roof.position.set(cx, h, cz);
   398→    this.scene.add(roof);
   399→  }

   401→  private buildProp(p: { kind: string; x: number; y: number; w: number; h: number; label?: string; rot?: number }): void {
   402→    const cx = p.x + p.w / 2;
   403→    const cz = p.y + p.h / 2;
   404→    const r = Math.min(p.w, p.h) / 2;
   405→    switch (p.kind) {
   406→      case 'tree': {
   407→        const trunk = new THREE.Mesh(
   408→          new THREE.CylinderGeometry(r * 0.16, r * 0.22, r * 1.1, 8),
   409→          new THREE.MeshLambertMaterial({ color: '#8a5a33' }),
   410→        );
   411→        trunk.position.set(cx, r * 0.55, cz);
   412→        trunk.castShadow = true;
   413→        const blob = new THREE.Mesh(
   414→          new THREE.IcosahedronGeometry(r * 0.92, 0),
   415→          new THREE.MeshLambertMaterial({ color: '#5FA84B' }),
   416→        );
   417→        blob.position.set(cx, r * 1.5, cz);
   418→        blob.castShadow = true;
   419→        const blob2 = new THREE.Mesh(
   420→          new THREE.IcosahedronGeometry(r * 0.55, 0),
   421→          new THREE.MeshLambertMaterial({ color: '#8AC926' }),
   422→        );
   423→        blob2.position.set(cx + r * 0.3, r * 1.9, cz - r * 0.2);
   424→        blob2.castShadow = true;
   425→        this.scene.add(trunk, blob, blob2);
   426→        break;
   427→      }
   428→      case 'trash': {
   429→        const bin = new THREE.Mesh(
   430→          new THREE.CylinderGeometry(r * 0.8, r * 0.66, r * 1.5, 10),
   431→          new THREE.MeshLambertMaterial({ color: '#4F772D' }),

   580→  /** به‌روزرسانی افکت‌ها (هر فریم، مستقل از گام فیزیک) */
   581→  update(dt: number): void {
   582→    // ذرات
   583→    let anyAlive = false;
   584→    let anyDead = false;
   585→    for (const p of this.particles) {
   586→      if (!p.active) continue;
   587→      p.life -= dt;
   588→      if (p.life <= 0) { p.active = false; anyDead = true; continue; }
   589→      p.x += p.vx * dt;
   590→      p.z += p.vz * dt;
   591→      p.y += p.vy * dt;
   592→      p.vy -= 560 * dt;
   593→      p.vx *= 0.94;
   594→      p.vz *= 0.94;
   595→      if (p.y < 12) { p.y = 12; p.vy *= -0.3; }
   596→      anyAlive = true;
   597→    }
   598→    if (anyDead) this.particles = this.particles.filter((p) => p.active);
   599→    // رد لاستیک
   600→    for (const s of this.skidData) {
   601→      if (s.life > 0) s.life -= dt;
   602→    }
   603→    // حباب‌ها
   604→    for (let i = this.bubbles.length - 1; i >= 0; i--) {
   605→      this.bubbles[i].ttl -= dt;
   606→      if (this.bubbles[i].ttl <= 0) this.bubbles.splice(i, 1);
   607→    }
   608→    if (anyAlive || anyDead) this.syncParticles();
   609→  }

   752→  private updateChaseCamera(car: Vehicle, dt: number, _t: number): void {
   753→    // نرم‌سازی زاویه (کوتاه‌ترین مسیر)
   754→    let diff = car.angle - this.camAngle;
   755→    while (diff > Math.PI) diff -= Math.PI * 2;
   756→    while (diff < -Math.PI) diff += Math.PI * 2;
   757→    this.camAngle += diff * Math.min(1, RENDER3D.camAngleLerp * dt);
   758→
   759→    const fx = Math.sin(this.camAngle);
   760→    const fz = -Math.cos(this.camAngle);
   761→    const back = RENDER3D.camBack + Math.abs(car.speed) * RENDER3D.camBackPerSpeed;
   762→    const desired = this.tmpP.set(car.x - fx * back, RENDER3D.camHeight, car.y - fz * back);
   763→    const k = 1 - Math.exp(-RENDER3D.camPosLerp * dt);
   764→    this.cam3.position.lerp(desired, k);
   765→
   766→    // لرزش از دوربین دوبعدی (برخورد)
   767→    const sh = this.camera.getShake();
   768→    this.cam3.position.x += sh.x * 0.9;
   769→    this.cam3.position.y += Math.abs(sh.y) * 0.5;
   770→    this.cam3.position.z += sh.y * 0.9;
   771→
   772→    // نگاه به جلوی ماشین
   773→    const look = this.tmpV.set(car.x + fx * 260, 60, car.y + fz * 260);
   774→    this.cam3.lookAt(look);
   775→  }

   777→  private syncParticles(): void {
   778→    let n = 0;
   779→    for (const p of this.particles) {
   780→      if (!p.active) continue;
   781→      const k = Math.max(0, p.life / p.maxLife);
   782→      this.tmpQ.identity();
   783→      this.tmpM.compose(this.tmpV.set(p.x, p.y, p.z), this.tmpQ, this.tmpS.setScalar(p.size * (0.5 + k * 0.5)));
   784→      this.particleMesh.setMatrixAt(n, this.tmpM);
   785→      this.particleMesh.setColorAt(n, p.color);
   786→      n++;
   787→      if (n >= RENDER3D.particleMax) break;
   788→    }
   789→    this.particleMesh.count = n;
   790→    this.particleMesh.instanceMatrix.needsUpdate = true;
   791→    if (this.particleMesh.instanceColor) this.particleMesh.instanceColor.needsUpdate = true;
   792→  }

   837→  /** مینی‌مپ کل شهر — نقشه جاده‌ها + بازیکن + ترافیک + هدف مأموریت */
   838→  private drawMinimap(ctx: CanvasRenderingContext2D, opts: Draw3DOptions): void {
   839→    const size = Math.min(RENDER3D.minimapSize, this.overlayW * 0.3);
   840→    const pad = 12;
   841→    const x0 = this.overlayW / 2 - size / 2;
   842→    const y0 = this.overlayH - size - pad;
   843→    const k = size / WORLD_SIZE;
   844→
   845→    ctx.save();
   846→    // قاب گرد نیمه‌شفاف
   847→    ctx.globalAlpha = 0.88;
   848→    const r = 16;
   849→    ctx.beginPath();
   850→    ctx.moveTo(x0 + r, y0);
   851→    ctx.arcTo(x0 + size, y0, x0 + size, y0 + size, r);
   852→    ctx.arcTo(x0 + size, y0 + size, x0, y0 + size, r);
   853→    ctx.arcTo(x0, y0 + size, x0, y0, r);
   854→    ctx.arcTo(x0, y0, x0 + size, y0, r);
   855→    ctx.closePath();
   856→    ctx.fillStyle = '#2e2622';
   857→    ctx.globalAlpha = 0.97;
   858→    ctx.fill();
   859→    ctx.globalAlpha = 1;
   860→    ctx.strokeStyle = '#F2B705';
   861→    ctx.lineWidth = 3;
   862→    ctx.stroke();
   863→    ctx.globalAlpha = 1;
   864→    ctx.clip(); // نقشه از قاب بیرون نزند
   865→
   866→    // جاده‌ها (۵ خط سفید — شبکه کامل ترافیک)
   867→    ctx.fillStyle = 'rgba(255,255,255,0.78)';
   868→    for (let i = 0; i < 5; i++) {
   869→      const c = (i * 1024 + 128) * k;
   870→      ctx.fillRect(x0 + c - 2.4, y0, 4.8, size);
   871→      ctx.fillRect(x0, y0 + c - 2.4, size, 4.8);
   872→    }
   873→

   916→  /** فلش جهت‌نمای هدف — روی اورلی، هم‌ظاهر نسخه دوبعدی */
   917→  private drawOffscreenArrow(
   918→    ctx: CanvasRenderingContext2D,
   919→    target: { x: number; y: number },
   920→    player: { x: number; y: number },
   921→  ): void {
   922→    const p = this.worldToScreen(target.x, target.y);
   923→    const W = this.overlayW;
   924→    const H = this.overlayH;
   925→    const margin = 84;
   926→    const sx = p.ok ? p.sx : -1e9;
   927→    const sy = p.ok ? p.sy : -1e9;
   928→    if (sx >= margin && sx <= W - margin && sy >= margin && sy <= H - margin) return;
   929→
   930→    const cx = Math.min(Math.max(sx, margin), W - margin);
   931→    const cy = Math.min(Math.max(sy, margin), H - margin);
   932→    const ang = Math.atan2(sy - cy, sx - cx);
   933→    const distM = Math.round(Math.hypot(target.x - player.x, target.y - player.y) / M.pxPerMeter);
   934→
   935→    ctx.save();
   936→    ctx.fillStyle = 'rgba(58,46,42,0.88)';
   937→    ctx.beginPath();
   938→    ctx.arc(cx, cy, 40, 0, Math.PI * 2);
   939→    ctx.fill();
   940→    ctx.strokeStyle = '#F2B705';
   941→    ctx.lineWidth = 4;
   942→    ctx.beginPath();
   943→    ctx.arc(cx, cy, 40, 0, Math.PI * 2);
   944→    ctx.stroke();
   945→    ctx.translate(cx, cy);
   946→    ctx.rotate(ang);
   947→    ctx.fillStyle = '#F2B705';
   948→    ctx.beginPath();
   949→    ctx.moveTo(22, 0);
   950→    ctx.lineTo(2, -13);
   951→    ctx.lineTo(2, -5);
   952→    ctx.lineTo(-18, -5);
   953→    ctx.lineTo(-18, 5);
   954→    ctx.lineTo(2, 5);
   955→    ctx.lineTo(2, 13);
   956→    ctx.closePath();
   957→    ctx.fill();
   958→    ctx.rotate(-ang);
   959→    ctx.fillStyle = '#FFF9EC';
   960→    ctx.font = '800 17px Vazirmatn, Tahoma, sans-serif';
   961→    ctx.textAlign = 'center';
   962→    ctx.textBaseline = 'middle';
   963→    ctx.fillText(`${faNum(distM)}م`, 0, 0);
   964→    ctx.restore();
   965→  }

   967→  /**
   968→   * حباب RTL با پرتره — نسخه صفحه‌ای (مختصات پیکسلی اورلی)
   969→   * هم‌ظاهر WorldRenderer.drawBubbleBox دوبعدی
   970→   */
   971→  private drawBubbleBox(
   972→    ctx: CanvasRenderingContext2D,
   973→    text: string,
   974→    anchorX: number,
   975→    anchorY: number,
   976→    anchorH = 100,
   977→    portraitId?: string,
   978→  ): void {
   979→    ctx.save();
   980→    ctx.direction = 'rtl';
   981→    ctx.font = '700 24px Vazirmatn, Tahoma, sans-serif';
   982→    ctx.textBaseline = 'middle';
   983→    ctx.textAlign = 'center';
   984→
   985→    const maxTextW = 260;
   986→    const words = text.split(' ');
   987→    const lines: string[] = [];
   988→    let current = '';
   989→    for (const w of words) {
   990→      const test = current ? `${current} ${w}` : w;
   991→      if (ctx.measureText(test).width > maxTextW && current) {
   992→        lines.push(current);
   993→        current = w;
   994→      } else current = test;
   995→    }
   996→    if (current) lines.push(current);
   997→
   998→    const lineH = 32;
   999→    const portraitR = 26;
  1000→    const bubbleW = Math.min(maxTextW + portraitR * 2 + 40, 360);
  1001→    const bubbleH = Math.max(72, lines.length * lineH + 34);
  1002→
  1003→    let bx = anchorX - bubbleW / 2;
  1004→    let by = anchorY - anchorH / 2 - bubbleH - 22;
  1005→    let tailBelow = false;
  1006→    if (by < 8) {
  1007→      by = anchorY + anchorH / 2 + 22;
  1008→      tailBelow = true;
  1009→    }
  1010→    if (by + bubbleH > this.overlayH - 8) by = this.overlayH - 8 - bubbleH;
  1011→    if (by < 8) by = 8;
  1012→    bx = Math.min(Math.max(bx, 8), this.overlayW - 8 - bubbleW);
  1013→    if (bx < 8) bx = 8;
  1014→
  1015→    // دم
  1016→    ctx.fillStyle = '#FFF9EC';
  1017→    ctx.strokeStyle = '#3A2E2A';
  1018→    ctx.lineWidth = 4;
  1019→    ctx.beginPath();
  1020→    if (tailBelow) {
  1021→      ctx.moveTo(anchorX - 13, by + 2);
  1022→      ctx.lineTo(anchorX, by - 20);
  1023→      ctx.lineTo(anchorX + 13, by + 2);
  1024→    } else {
  1025→      ctx.moveTo(anchorX - 13, by + bubbleH - 2);
  1026→      ctx.lineTo(anchorX, by + bubbleH + 20);
  1027→      ctx.lineTo(anchorX + 13, by + bubbleH - 2);
  1028→    }
  1029→    ctx.closePath();
  1030→    ctx.fill();
  1031→    ctx.stroke();
  1032→
  1033→    // بدنه
  1034→    ctx.beginPath();
  1035→    const r = 18;
  1036→    ctx.moveTo(bx + r, by);
  1037→    ctx.arcTo(bx + bubbleW, by, bx + bubbleW, by + bubbleH, r);
  1038→    ctx.arcTo(bx + bubbleW, by + bubbleH, bx, by + bubbleH, r);
  1039→    ctx.arcTo(bx, by + bubbleH, bx, by, r);
  1040→    ctx.arcTo(bx, by, bx + bubbleW, by, r);
  1041→    ctx.closePath();
  1042→    ctx.fill();
  1043→    ctx.stroke();
  1044→
  1045→    // پرتره دایره‌ای
  1046→    const pid = portraitId ?? 'char_driver_01_idle';
  1047→    const pcx = bx + bubbleW - portraitR - 14;
  1048→    const pcy = by + bubbleH / 2;
  1049→    const asset = this.assets.get(pid);
  1050→    const img = asset instanceof AudioBuffer ? undefined : asset;
  1051→    if (img) {
  1052→      ctx.save();
  1053→      ctx.beginPath();
  1054→      ctx.arc(pcx, pcy, portraitR, 0, Math.PI * 2);
  1055→      ctx.clip();
  1056→      ctx.drawImage(img, pcx - portraitR, pcy - portraitR, portraitR * 2, portraitR * 2);
  1057→      ctx.restore();
  1058→    } else {
  1059→      ctx.fillStyle = '#E8C39E';
  1060→      ctx.beginPath();
  1061→      ctx.arc(pcx, pcy, portraitR, 0, Math.PI * 2);
  1062→      ctx.fill();
  1063→    }
  1064→    ctx.strokeStyle = '#3A2E2A';
  1065→    ctx.lineWidth = 3.5;
  1066→    ctx.beginPath();
  1067→    ctx.arc(pcx, pcy, portraitR, 0, Math.PI * 2);
  1068→    ctx.stroke();
  1069→
  1070→    // متن
  1071→    ctx.fillStyle = '#3A2E2A';
  1072→    const textAreaCenter = bx + (bubbleW - portraitR * 2 - 18) / 2;
  1073→    lines.forEach((ln, i) => {
  1074→      ctx.fillText(ln, textAreaCenter, by + bubbleH / 2 + (i - (lines.length - 1) / 2) * lineH);
  1075→    });
  1076→    ctx.restore();
  1077→  }

  1081→  /** w/h پیکسل فیزیکی کانواس WebGL + dpr برای اورلی */
  1082→  resize(pixelW: number, pixelH: number, dpr: number): void {
  1083→    this.dpr = dpr;
  1084→    this.r3.setPixelRatio(1);
  1085→    this.r3.setSize(pixelW, pixelH, false);
  1086→    this.overlay.width = pixelW;
  1087→    this.overlay.height = pixelH;
  1088→    // اندازه CSS اورلی = ۱۰۰٪ والد (در DOM تنظیم شده)
  1089→    // ابعاد منطقی برای پروجکشن = پیکسل CSS
  1090→    const rect = this.overlay.getBoundingClientRect();
  1091→    this.overlayW = rect.width || pixelW / dpr;
  1092→    this.overlayH = rect.height || pixelH / dpr;
  1093→    // FOV تطبیقی: در صفحه‌های باریک (موبایل عمودی) زاویه دید بازتر می‌شود
  1094→    const aspect = this.overlayW / Math.max(1, this.overlayH);
  1095→    const extraFov = Math.min(26, Math.max(0, (1 / Math.max(aspect, 0.2) - 1) * 22));
  1096→    this.cam3.fov = RENDER3D.fov + extraFov;
  1097→    this.cam3.aspect = aspect;
  1098→    this.cam3.updateProjectionMatrix();
  1099→  }

  1101→  dispose(): void {
  1102→    this.scene.traverse((obj) => {
  1103→      const mesh = obj as THREE.Mesh;
  1104→      if (mesh.geometry) mesh.geometry.dispose();
  1105→      const mat = (mesh as unknown as { material?: THREE.Material | THREE.Material[] }).material;
  1106→      if (Array.isArray(mat)) mat.forEach((m) => m.dispose());
  1107→      else if (mat) mat.dispose();
  1108→    });
  1109→    // توجه: کانتکت WebGL را نمی‌بندیم (loseContext) — چون کانواس در
  1110→    // StrictMode/HMR دوباره استفاده می‌شود و کانتکت همان است.
  1111→    this.r3.dispose();
  1112→    this.overlay.parentElement?.removeChild(this.overlay);
  1113→  }
