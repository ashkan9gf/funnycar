    50→/** پنل تنظیمات صدا (در منو و Pause استفاده می‌شود) */
    51→function SettingsPanel({ game, onClose }: { game: GameInstance; onClose?: () => void }) {
    52→  const [s, setS] = useState<Settings>(game.getSettings());
    53→  const rows: Array<{ key: keyof Settings; label: string; icon: React.ReactNode }> = [
    54→    { key: "music", label: "موسیقی", icon: <Music className="h-5 w-5" /> },
    55→    { key: "sfx", label: "افکت‌ها", icon: <Volume2 className="h-5 w-5" /> },
    56→    { key: "voice", label: "جملات طنز", icon: <Mic className="h-5 w-5" /> },
    57→  ];
    58→  return (
    59→    <div className="w-full space-y-3">
    60→      {rows.map((r) => (
    61→        <div key={r.key} className="flex items-center gap-3">
    62→          <span className="text-[#3a2e2a]">{r.icon}</span>
    63→          <span className="w-24 text-sm font-bold text-[#3a2e2a]">{r.label}</span>
    64→          <input
    65→            type="range"
    66→            className="fc-range flex-1"
    67→            min={0}
    68→            max={1}
    69→            step={0.05}
    70→            value={s[r.key]}
    71→            aria-label={`صدای ${r.label}`}
    72→            onChange={(e) => {
    73→              const v = Number(e.target.value);
    74→              setS((p) => ({ ...p, [r.key]: v }));
    75→              game.setSetting(r.key, v);
    76→            }}
    77→          />
    78→          <span className="w-10 text-center text-sm font-bold text-[#3a2e2a]">{fa(Math.round(s[r.key] * 100))}٪</span>
    79→        </div>
    80→      ))}
    81→      <div className="flex items-center gap-3 opacity-50">
    82→        <span className="text-sm font-bold text-[#3a2e2a]">لرزش گوشی</span>
    83→        <span className="text-xs text-[#3a2e2a]">(فازهای بعدی)</span>
    84→      </div>
    85→      {onClose && (
    86→        <button
    87→          type="button"
    88→          className="fc-btn mt-2 w-full rounded-2xl bg-[#3a2e2a] py-2.5 text-white"
    89→          onPointerDown={(e) => {
    90→            e.preventDefault();
    91→            onClose();
    92→          }}
    93→        >
    94→          بستن
    95→        </button>
    96→      )}
    97→    </div>
    98→  );
    99→}

   118→export default function GameShell() {
   119→  const canvasRef = useRef<HTMLCanvasElement>(null);
   120→  const gameRef = useRef<GameInstance | null>(null);
   121→  const steerRef = useRef({ left: false, right: false });
   122→  const toastTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   123→

   125→  const [game, setGame] = useState<GameInstance | null>(null);
   126→
   127→  const [phase, setPhase] = useState<GameState | "boot">("boot");
   128→  const [bootProgress, setBootProgress] = useState(0);
   129→  const [hud, setHud] = useState<HudData>({ speedKmh: 0, coins: 0, reputation: 50, fps: 60, showFps: false, placeholders: 0, mission: null, health: 100, engineDead: false });

   131→  const [showSettings, setShowSettings] = useState(false);
   132→  const [showMissions, setShowMissions] = useState(false);
   133→  /** فیلتر دسته مأموریت (فاز ۵) — null = همه (مأموریت‌های اصلی) */
   134→  const [missionFilter, setMissionFilter] = useState<MissionCategory | null>(null);
   135→  const [missions, setMissions] = useState<MissionListItem[]>([]);
   136→  const [missionResult, setMissionResult] = useState<MissionResultData | null>(null);
   137→
   138→  const showToast = useCallback((msg: string) => {
   139→    setToast(msg);
   140→    if (toastTimer.current) clearTimeout(toastTimer.current);
   141→    toastTimer.current = setTimeout(() => setToast(null), 1800);
   142→  }, []);
   143→
   144→  // ---------- راه‌اندازی موتور ----------
   145→  useEffect(() => {
   146→    let cancelled = false;
   147→    const canvas = canvasRef.current;
   148→    if (!canvas) return;
   149→    createGame(
   150→      canvas,
   151→      {
   152→        onStateChange: (s) => setPhase(s),
   153→        onHud: (h) => setHud(h),
   154→        onMissionEnd: (r) => setMissionResult(r),
   155→      },
   156→      (done, total) => setBootProgress(Math.round((done / total) * 100)),

   159→        inst.dispose();
   160→        return;
   161→      }
   162→      gameRef.current = inst;
   163→      setGame(inst);
   164→    });
   165→    return () => {
   166→      cancelled = true;
   167→      gameRef.current?.dispose();
   168→      gameRef.current = null;
   169→    };
   170→  }, []);
   171→
   172→  // ---------- فرمان ترکیبی (چپ+راست هم‌زمان = مستقیم) ----------

   175→    gameRef.current?.setVirtualSteer(left === right ? 0 : right ? 1 : -1);
   176→  }, []);
   177→  const steerLeft = useCallback(
   178→    (down: boolean) => {
   179→      steerRef.current.left = down;
   180→      applySteer();
   181→    },
   182→    [applySteer],
   183→  );
   184→  const steerRight = useCallback(
   185→    (down: boolean) => {
   186→      steerRef.current.right = down;
   187→      applySteer();
   188→    },
   189→    [applySteer],
   190→  );
   191→
   192→  const repColor = hud.reputation >= 70 ? "#8AC926" : hud.reputation >= 35 ? "#F2B705" : "#D64545";

   199→      <canvas ref={canvasRef} className="fc-canvas absolute inset-0 h-full w-full" />
   200→
   201→      {/* ============ صفحه بوت ============ */}
   202→      {phase === "boot" && (
   203→        <div className="absolute inset-0 z-50 flex items-center justify-center bg-[#2A9D8F]">
   204→          <div className="fc-panel fc-pop flex flex-col items-center gap-4 bg-[#FFF9EC] px-10 py-8">
   205→            <Loader2 className="h-10 w-10 animate-spin text-[#2A9D8F]" />
   206→            <p className="text-xl font-extrabold text-[#3a2e2a]">در حال باز کردن گاراژ...</p>
   207→            <div className="h-4 w-56 overflow-hidden rounded-full border-4 border-[#3a2e2a] bg-[#FFF3D6]">
   208→              <div className="h-full bg-[#8AC926] transition-all" style={{ width: `${bootProgress}%` }} />
   209→            </div>
   210→            <p className="text-sm font-bold text-[#3a2e2a]">٪{fa(bootProgress)}</p>
   211→          </div>
   212→        </div>
   213→      )}
   214→
   215→      {/* ============ منوی اصلی ============ */}
   216→      {phase === "menu" && (
   217→        <div className="absolute inset-0 z-30 flex flex-col overflow-y-auto bg-gradient-to-b from-[#3a2e2a]/10 via-transparent to-[#3a2e2a]/55">
   218→          <div className="fc-pop mx-auto flex w-full max-w-md flex-1 flex-col items-center justify-center gap-5 px-6 py-8">
   219→            {/* لوگو */}
   220→            <div className="flex flex-col items-center gap-2">
   221→              <div className="fc-btn flex h-20 w-20 items-center justify-center rounded-3xl bg-[#F2B705]">
   222→                <Car className="h-11 w-11 text-[#3a2e2a]" strokeWidth={2.4} />
   223→              </div>
   224→              <h1 className="mt-2 text-5xl font-extrabold text-white drop-shadow-[0_4px_0_rgba(58,46,42,0.9)]">فانی‌کار</h1>
   225→              <p className="text-sm font-bold text-white/90 drop-shadow-[0_2px_0_rgba(58,46,42,0.7)]">
   226→                آبرو می‌بری یا می‌ریزی؟
   227→              </p>
   228→            </div>
   229→
   230→            {/* آمار بازیکن */}
   231→            <div className="flex items-center gap-3">
   232→              <div className="fc-panel flex items-center gap-2 bg-[#FFF9EC] px-4 py-2">
   233→                <Coins className="h-5 w-5 text-[#B8801A]" />
   234→                <span className="font-extrabold text-[#3a2e2a]">{fa(hud.coins)}</span>
   235→              </div>

   242→            {/* دکمه اصلی */}
   243→            <button
   244→              type="button"
   245→              className="fc-btn w-full max-w-xs rounded-3xl bg-[#8AC926] py-4 text-2xl font-extrabold text-[#3a2e2a]"

   247→                e.preventDefault();

   249→              }}

   251→              رانندگی آزاد
   252→            </button>
   253→
   254→            {/* حالت‌های بازی — فاز ۵: همه فعال! */}
   255→            <div className="grid w-full max-w-xs grid-cols-2 gap-3">
   256→              {([
   257→                { cat: "story" as MissionCategory, icon: <Map className="h-6 w-6" />, label: "مأموریت‌ها" },
   258→                { cat: "park" as MissionCategory, icon: <Target className="h-6 w-6" />, label: "پارک دوبل" },
   259→                { cat: "taxi" as MissionCategory, icon: <Users className="h-6 w-6" />, label: "مسافرکشی" },
   260→                { cat: "challenge" as MissionCategory, icon: <PartyPopper className="h-6 w-6" />, label: "چالش‌ها" },
   261→              ]).map((m) => (
   262→                <button
   263→                  key={m.cat}
   264→                  type="button"
   265→                  className="fc-btn flex flex-col items-center gap-1 rounded-2xl bg-[#8AC926] py-3 text-[#3a2e2a]"
   266→                  onPointerDown={(e) => {
   267→                    e.preventDefault();
   268→                    setMissionFilter(m.cat);
   269→                    setMissions(gameRef.current?.listMissions(m.cat) ?? []);
   270→                    setShowMissions(true);
   271→                  }}
   272→                >
   273→                  {m.icon}
   274→                  <span className="text-sm font-extrabold">{m.label}</span>
   275→                  <span className="text-[10px] font-bold opacity-60">فاز ۵ ✓</span>

   278→            </div>
   279→
   280→            {/* تنظیمات */}
   281→            {showSettings ? (
   282→              <div className="fc-panel fc-pop w-full max-w-xs bg-[#FFF9EC] p-4">
   283→                <div className="mb-3 flex items-center justify-between">
   284→                  <span className="text-lg font-extrabold text-[#3a2e2a]">تنظیمات</span>

   287→                  </TapButton>

   290→              </div>
   291→            ) : (
   292→              <button
   293→                type="button"
   294→                className="fc-btn flex items-center gap-2 rounded-2xl bg-[#FFF9EC] px-6 py-2.5 font-extrabold text-[#3a2e2a]"
   295→                onPointerDown={(e) => {
   296→                  e.preventDefault();
   297→                  setShowSettings(true);
   298→                }}
   299→              >
   300→                <Settings className="h-5 w-5" />
   301→                تنظیمات
   302→              </button>
   303→            )}
   304→
   305→            {/* یادداشت Placeholder (شفافیت — قانون بخش ۳۲ سند مادر) */}
   306→            {hud.placeholders > 0 && (
   307→              <p className="max-w-xs text-center text-[11px] font-bold leading-5 text-white/80">
   308→                {fa(hud.placeholders)} فایل موقت (Placeholder) فعال است — طبق مستندات
   309→                «ASSET_HANDBOOK» و «ASSET_INVENTORY» به‌تدریج با Assetهای واقعی جایگزین می‌شود.
   310→              </p>
   311→            )}
   312→
   313→            <p className="mt-auto text-center text-[11px] font-bold leading-5 text-white/70">
   314→              دسکتاپ: جهت‌نما / WASD رانندگی — H بوق — P توقف — F شمارنده FPS

   316→              نسخه فاز ۱ • موتور اختصاصی HTML5
   317→            </p>
   318→          </div>
   319→        </div>

   323→      {phase === "menu" && showMissions && (

   333→                className="h-11 w-11 bg-[#FFF9EC] text-[#3a2e2a]"

   339→            {missions.map((m) => (
   340→              <button
   341→                key={m.id}
   342→                type="button"
   343→                className="fc-btn fc-panel flex w-full flex-col items-start gap-1 bg-[#FFF9EC] p-4 text-right"

   345→                  e.preventDefault();
   346→                  gameRef.current?.startMission(m.id);

   350→                <div className="flex w-full items-center justify-between">
   351→                  <span className="text-lg font-extrabold text-[#3a2e2a]">{m.title}</span>

   373→              {CATEGORY_FA[missionFilter ?? "story"].hint}

   379→      {/* ============ HUD گیم‌پلی ============ */}

   384→            <div className="pointer-events-auto">
   385→              <TapButton ariaLabel="توقف بازی" className="h-12 w-12 bg-[#FFF9EC] text-[#3a2e2a]" onTap={() => gameRef.current?.pause()}>
   386→                <Pause className="h-6 w-6" />
   387→              </TapButton>

   393→                  <div className="flex items-center justify-between gap-2">
