    48→/** پنل تنظیمات صدا (در منو و Pause استفاده می‌شود) */
    49→function SettingsPanel({ game, onClose }: { game: GameInstance; onClose?: () => void }) {
    50→  const [s, setS] = useState<Settings>(game.getSettings());
    51→  const rows: Array<{ key: keyof Settings; label: string; icon: React.ReactNode }> = [
    52→    { key: "music", label: "موسیقی", icon: <Music className="h-5 w-5" /> },
    53→    { key: "sfx", label: "افکت‌ها", icon: <Volume2 className="h-5 w-5" /> },
    54→    { key: "voice", label: "جملات طنز", icon: <Mic className="h-5 w-5" /> },
    55→  ];
    56→  return (
    57→    <div className="w-full space-y-3">
    58→      {rows.map((r) => (
    59→        <div key={r.key} className="flex items-center gap-3">
    60→          <span className="text-[#3a2e2a]">{r.icon}</span>
    61→          <span className="w-24 text-sm font-bold text-[#3a2e2a]">{r.label}</span>
    62→          <input
    63→            type="range"
    64→            className="fc-range flex-1"
    65→            min={0}
    66→            max={1}
    67→            step={0.05}
    68→            value={s[r.key]}
    69→            aria-label={`صدای ${r.label}`}
    70→            onChange={(e) => {
    71→              const v = Number(e.target.value);
    72→              setS((p) => ({ ...p, [r.key]: v }));
    73→              game.setSetting(r.key, v);
    74→            }}
    75→          />
    76→          <span className="w-10 text-center text-sm font-bold text-[#3a2e2a]">{fa(Math.round(s[r.key] * 100))}٪</span>
    77→        </div>
    78→      ))}
    79→      <div className="flex items-center gap-3 opacity-50">
    80→        <span className="text-sm font-bold text-[#3a2e2a]">لرزش گوشی</span>
    81→        <span className="text-xs text-[#3a2e2a]">(فازهای بعدی)</span>
    82→      </div>
    83→      {onClose && (
    84→        <button
    85→          type="button"
    86→          className="fc-btn mt-2 w-full rounded-2xl bg-[#3a2e2a] py-2.5 text-white"
    87→          onPointerDown={(e) => {
    88→            e.preventDefault();
    89→            onClose();
    90→          }}
    91→        >
    92→          بستن
    93→        </button>
    94→      )}
    95→    </div>
    96→  );
    97→}

   129→  }, []);
   130→

   132→  useEffect(() => {

   188→      {/* ============ صفحه بوت ============ */}

   210→              </div>

   290→              <button

   293→                onPointerDown={(e) => {
   294→                  e.preventDefault();
   295→                  setShowSettings(true);

   297→              >
   298→                <Settings className="h-5 w-5" />
   299→                تنظیمات

   301→            )}
   302→
   303→            {/* یادداشت Placeholder (شفافیت — قانون بخش ۳۲ سند مادر) */}

   305→              <p className="max-w-xs text-center text-[11px] font-bold leading-5 text-white/80">

   311→            <p className="mt-auto text-center text-[11px] font-bold leading-5 text-white/70">
