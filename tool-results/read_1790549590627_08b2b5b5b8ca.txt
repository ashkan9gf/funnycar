   313→  private onPotholeHit(x: number, y: number): void {
   314→    this.camera.addShake(CAM.maxShake * 0.55);
   315→    this.audio.crash(false);
   316→    this.vehicle.speed *= MISSIONS.potholeSlowFactor;

   321→  /** پایان مأموریت — جایزه/جریمه + ذخیره + رویداد به پوسته */
   322→  private finishMission(
   323→    success: boolean,
   324→    missionId: string,
   325→    baseCoins: number,
   326→    baseRep: number,
   327→    reason: string | null,
   328→    _stats: unknown,
   329→  ): void {
   330→    if (this.state !== 'playing') return;
   331→    const def = this.mission.getDef();
   332→    const title = def?.title ?? missionId;
   333→
   334→    let coins = 0;
   335→    let rep = 0;
   336→    if (success) {
   337→      // بونوس آبروی بالا (سند بخش ۷: ×1.25 وقتی آبرو ≥ ۸۰)
   338→      const repNow = this.save.get().reputation;
   339→      coins = Math.round(baseCoins * (repNow >= MISSIONS.bonusRepThreshold ? MISSIONS.bonusFactor : 1));
   340→      rep = baseRep;
   341→      this.save.mutate((d) => {
   342→        d.coins += coins;
   343→        d.reputation = Math.min(REPUTATION.max, d.reputation + rep);
   344→        if (!d.completedMissions.includes(missionId)) d.completedMissions.push(missionId);
   345→      });
   346→    }
   347→
   348→    this.audio.coin();
   349→    if (success) this.fx.spawnBurst(this.vehicle.x, this.vehicle.y - 30, '#F2B705', 16, 150);
   350→    this.voice.trigger(success ? 'mission_success' : 'mission_fail', performance.now());
   351→
   352→    const line = this.pickLine(success ? 'mission_success' : 'mission_fail');
   353→    this.callbacks.onMissionEnd?.({ success, missionId, title, coins, rep, reason, line });
   354→  }

   370→  /** رویداد تصادفی فعال شد → صدا + دیالوگ */
   371→  private onRandomEvent(kind: 'cat' | 'cones' | 'taxi', x: number, y: number): void {
   372→    if (this.state !== 'playing') return;
   373→    const nowMs = performance.now();
   374→    if (kind === 'cat') {
   375→      this.voice.trigger('cat_event', nowMs);
   376→    } else if (kind === 'cones') {
   377→      this.voice.trigger('cones_ahead', nowMs);
   378→    } else if (kind === 'taxi') {
   379→      const taxi = this.randomEvents.nearestTaxi(this.traffic.cars, x, y);
   380→      if (taxi) {
   381→        this.audio.horn(Math.floor(Math.random() * 3));
   382→        this.fx.pushBubble(this.pickLine('traffic_honk'), taxi.x, taxi.y, 'char_driver_01_idle', 2.6);
   383→      }
   384→    }
   385→  }

   387→  /** برخورد بازیکن با خودروی ترافیک — بازیکن کنار می‌رود، ترافیک مکث می‌کند + آسیب (فاز BIG-IR) */
   388→  private resolveTrafficCollisions(dt: number): void {
   389→    const v = this.vehicle;
   390→    this.crashCooldown = Math.max(0, this.crashCooldown - dt);
   391→    for (const c of this.traffic.cars) {
   392→      const hw = v.hitW / 2 + c.hitW / 2;
   393→      const hh = v.hitH / 2 + c.hitH / 2;
   394→      const dx = v.x - c.x;
   395→      const dy = v.y - c.y;
   396→      const penX = hw - Math.abs(dx);
   397→      const penY = hh - Math.abs(dy);
   398→      if (penX <= 0 || penY <= 0) continue;
   399→
   400→      const impact = Math.hypot(v.vx, v.vy);
   401→      if (penX < penY) v.x += dx >= 0 ? penX : -penX;
   402→      else v.y += dy >= 0 ? penY : -penY;
   403→      v.speed *= 0.5;
   404→      v.latVel *= 0.4;
   405→      c.stun(TRAFFIC.stunnedSec);
   406→
   407→      if (this.crashCooldown <= 0 && impact > COLLISION.softCrashSpeed) {
   408→        this.crashCooldown = 0.8;
   409→        // برخورد با ترافیک هم برای مأموریت شمرده می‌شود (فاز 4)
   410→        this.mission.onCrash(impact > COLLISION.hardCrashSpeed ? 'hard' : 'soft');
   411→        this.audio.crash(false);
   412→        this.camera.addShake(CAM.maxShake * 0.5);
   413→        this.fx.spawnBurst(v.x, v.y, '#B9B9B9', 10);
   414→        // آسیب برخورد با خودرو (مانع متحرک بزرگ) — جلو/عقب با جهت
   415→        const dmg = Math.min(DAMAGE.trafficHitMax, DAMAGE.trafficHitBase + impact * DAMAGE.trafficHitScale);
   416→        const fwd = Math.sin(v.angle) * (c.x - v.x) - Math.cos(v.angle) * (c.y - v.y);
   417→        const front = fwd > 0;
   418→        v.applyDamage(dmg, front);
   419→        this.renderer3d?.onPlayerDamaged(dmg, front, (v.x + c.x) / 2, (v.y + c.y) / 2);
   420→        this.save.mutate((d) => {
   421→          d.reputation = Math.max(REPUTATION.min, d.reputation - REPUTATION.trafficCrashPenalty);
   422→        });
   423→        this.voice.trigger('car_crash', performance.now());
   424→        if (Math.random() < 0.5) {
   425→          this.fx.pushBubble(this.pickLine('traffic_honk'), c.x, c.y, 'char_driver_01_idle', 2.4);
   426→        }
   427→      }
   428→      break; // یک برخورد در هر گام کافی است
   429→    }
   430→  }

   443→  /** مانع‌های کوچک با سرعت می‌شکنند + بعد از مدت‌هایی دوباره می‌آیند */
   444→  private updateBreakables(dt: number): void {
   445→    this.breakCooldown = Math.max(0, this.breakCooldown - dt);
   446→    const v = this.vehicle;
   447→
   448→    // تایمر بازگشت شکسته‌ها
   449→    for (const p of this.world.props) {
   450→      if (p.broken) {
   451→        p.brokenTimer = (p.brokenTimer ?? 0) - dt;
   452→        if (p.brokenTimer <= 0) p.broken = false;
   453→      }
   454→    }
   455→
   456→    if (this.breakCooldown > 0 || Math.abs(v.speed) < BREAKABLE.breakSpeed) return;
   457→    const hw = v.hitW / 2;
   458→    const hh = v.hitH / 2;
   459→
   460→    for (const p of this.world.props) {
   461→      if (!p.breakable || p.broken) continue;
   462→      // برخورد دایره ماشین با مستطیل مانع
   463→      const cx = Math.max(p.x, Math.min(v.x, p.x + p.w));
   464→      const cy = Math.max(p.y, Math.min(v.y, p.y + p.h));
   465→      const dx = v.x - cx;
   466→      const dy = v.y - cy;
   467→      if (dx * dx + dy * dy > hw * hw) continue;
   468→
   469→      // شکستن!
   470→      p.broken = true;
   471→      p.brokenTimer = BREAKABLE.respawnSec;
   472→      this.breakCooldown = BREAKABLE.breakCooldownSec;
   473→      const color = GameManager.BREAK_COLORS[p.kind] ?? '#9b9b9b';
   474→      this.fx.spawnBurst(p.x + p.w / 2, p.y + p.h / 2, color, 12, 230);
   475→      this.audio.bump();
   476→      this.camera.addShake(3.2);
   477→      v.speed *= BREAKABLE.speedLoss;
   478→      // گاهی یه سکه از وسطش می‌پرد!
   479→      if (Math.random() < BREAKABLE.coinChance) {
   480→        this.save.mutate((d) => { d.coins += 1; });
   481→        this.fx.spawnBurst(p.x + p.w / 2, p.y + p.h / 2, '#F2B705', 5, 140);
   482→      }
   483→      break; // یک شکستن در هر گام
   484→    }
   485→
   486→    // مخروط‌های رویداد هم با سرعت می‌شکنند (تکنیکی — همان سیستم)
   487→    const cones = this.randomEvents.cones;
   488→    for (let i = cones.length - 1; i >= 0; i--) {
   489→      const c = cones[i];
   490→      const dx = v.x - c.x;
   491→      const dy = v.y - c.y;
   492→      if (dx * dx + dy * dy < 60 * 60) {
   493→        cones.splice(i, 1);
   494→        this.fx.spawnBurst(c.x, c.y, '#F4A261', 8, 200);
   495→        this.audio.bump();
   496→        this.camera.addShake(2.4);
   497→      }
   498→    }
   499→  }

   511→  /** زون تعمیرگاه: ترمیم تدریجی داخل شعاع */
   512→  private updateRepairZones(dt: number, nowMs: number): void {
   513→    const v = this.vehicle;
   514→    if (v.health >= DAMAGE.maxHealth) {
   515→      this.repairVoiceLatch = false;
   516→      return;
   517→    }
   518→    for (const z of this.world.repairZones) {
   519→      const d = Math.hypot(v.x - z.x, v.y - z.y);
   520→      if (d > DAMAGE.repairRadius) continue;
   521→
   522→      v.health = Math.min(DAMAGE.maxHealth, v.health + DAMAGE.repairRate * dt);
   523→      if (Math.random() < 0.25) {
   524→        this.fx.spawnBurst(v.x + (Math.random() - 0.5) * 60, v.y + (Math.random() - 0.5) * 60, '#2A9D8F', 2, 90);
   525→      }
   526→      if (v.health >= DAMAGE.maxHealth) {
   527→        v.fullRepair();
   528→        this.engineDeadShown = false;
   529→        this.renderer3d?.resetPlayerDamage();
   530→        this.audio.coin();
   531→        this.fx.spawnBurst(v.x, v.y, '#2A9D8F', 14, 160);
   532→        if (!this.repairVoiceLatch) {
   533→          this.repairVoiceLatch = true;
   534→          this.voice.trigger('repair_done', nowMs);
   535→        }
   536→      }
   537→      break;
   538→    }
   539→  }

   542→  towToRepair(): void {

   545→    const coins = this.save.get().coins;

   570→  // ============ حلقه ============
   571→  private loop = (now: number): void => {
   572→    // اگر نمونه دیگری جای این را گرفته، خودکشی کن (ضد زامبی)
   573→    if (this.disposed || GameManager.live !== this) return;
   574→    this.raf = requestAnimationFrame(this.loop);
   575→
   576→    try {
   577→      const rawDt = Math.min(0.1, (now - this.lastFrame) / 1000 || 0);
   578→      this.lastFrame = now;
   579→
   580→      // FPS
   581→      this.fpsAccum += rawDt;
   582→      this.fpsCount++;
   583→      if (this.fpsAccum >= 0.5) {
   584→        this.fpsValue = Math.round(this.fpsCount / this.fpsAccum);
   585→        this.fpsAccum = 0;
   586→        this.fpsCount = 0;
   587→      }
   588→
   589→      if (this.state === 'playing') {
   590→        this.accumulator += rawDt;
   591→        let steps = 0;
   592→        while (this.accumulator >= FIXED_DT && steps < 5) {
   593→          this.step(FIXED_DT);
   594→          this.accumulator -= FIXED_DT;
   595→          steps++;
   596→        }
   597→        if (steps >= 5) this.accumulator = 0; // جلوگیری از Spiral of Death
   598→        this.voice.update(rawDt);
   599→      }
   600→
   601→      this.fx.update(rawDt);
   602→      this.draw();
   603→      this.emitHud(rawDt);
   604→    } catch (err) {
   605→      // خطای یک فریم هرگز بازی را نمی‌بندد
   606→      console.warn('[Game] frame error:', err);
   607→      this.accumulator = 0;
   608→    }
   609→  };

   611→  private step(dt: number): void {
   612→    let input = this.input.collect();
   613→
   614→    // بوق از کیبورد (H)
   615→    if (input.hornPressed) this.pressHorn();
   616→
   617→    // ---------- موتور خاموش: گاز کار نمی‌کند (فاز BIG-IR) ----------
   618→    if (this.vehicle.dead) {
   619→      input = { ...input, gas: false };
   620→    }
   621→
   622→    const result = this.vehicle.update(dt, input, this.world.colliders);
   623→    const nowMs = performance.now();
   624→
   625→    // ---------- برخورد با موانع ثابت ----------
   626→    this.crashCooldown = Math.max(0, this.crashCooldown - dt);
   627→    if (result.crashed && this.crashCooldown <= 0) {
   628→      const hard = result.impactSpeed > 260;
   629→      // گزارش برخورد به مأموریت فقط یک‌بار در هر رویداد (نه هر فریم)
   630→      this.mission.onCrash(hard ? 'hard' : 'soft');
   631→      this.crashCooldown = 0.8;
   632→      this.audio.crash(hard);
   633→      this.camera.addShake(hard ? CAM.maxShake : CAM.maxShake * 0.45);
   634→      // نقطه برخورد داینامیک: ذرات + جرقه در نقطه تماس + خرابی بدنه (فاز BIG-IR)
   635→      this.fx.spawnBurst(result.impactX, result.impactY, hard ? '#9B9B9B' : '#D6D6D6', hard ? 14 : 8);
   636→      this.renderer3d?.onPlayerDamaged(result.damage, result.front, result.impactX, result.impactY);
   637→      this.save.mutate((d) => {
   638→        d.reputation = Math.max(REPUTATION.min, d.reputation - (hard ? REPUTATION.hardCrashPenalty : REPUTATION.softCrashPenalty));
   639→      });
   640→      this.voice.trigger('car_crash', nowMs);
   641→    }
   642→
   643→    // ---------- ترمز ----------
   644→    if (input.brake && Math.abs(this.vehicle.speed) > 160) {
   645→      if (!this.brakeSoundLatch) {
   646→        this.audio.brake();
   647→        this.brakeSoundLatch = true;
   648→        this.voice.trigger('hard_brake', nowMs);
   649→      }
   650→    } else if (!input.brake) {
   651→      this.brakeSoundLatch = false;
   652→    }
   653→
   654→    // ---------- درفت ملایم (فاز 2): رد لاستیک + صدای سایش + دود ----------
   655→    if (result.skidding) {
   656→      // نقطه محور عقب (کمی پشت مرکز ماشین)
   657→      const back = this.vehicle.drawH * 0.32;
   658→      const ax = this.vehicle.x - Math.sin(this.vehicle.angle) * back;
   659→      const ay = this.vehicle.y + Math.cos(this.vehicle.angle) * back;
   660→      this.fx.addSkidMark(ax, ay, this.vehicle.angle);
   661→      this.skidSoundTimer -= dt;
   662→      if (this.skidSoundTimer <= 0) {
   663→        this.skidSoundTimer = DRIFT.skidSoundInterval;
   664→        const intensity = Math.min(1, Math.abs(this.vehicle.latVel) / DRIFT.maxLateral + 0.35);
   665→        this.audio.skid(intensity);
   666→        this.fx.spawnBurst(ax, ay, 'rgba(160,160,160,0.9)', 2, 60);
   667→      }
   668→      this.voice.trigger('drift', nowMs);
   669→    } else {
   670→      this.skidSoundTimer = 0;
   671→    }
   672→
   673→    // ---------- سرعت‌گیرها (فاز GTA-IR): «تق» + کندی + لرزش + غبار ----------
   674→    this.bumpCooldown = Math.max(0, this.bumpCooldown - dt);
   675→    if (this.bumpCooldown <= 0 && Math.abs(this.vehicle.speed) > RENDER3D.bumpMinSpeed) {
   676→      for (const sb of this.world.speedBumps) {
   677→        const dx = this.vehicle.x - sb.x;
   678→        const dy = this.vehicle.y - sb.y;
   679→        // باند عبور عرض جاده: جاده افقی = عبور در راستای X
   680→        const inside = sb.horiz
   681→          ? Math.abs(dx) < 24 && Math.abs(dy) < 100
   682→          : Math.abs(dy) < 24 && Math.abs(dx) < 100;
   683→        if (inside) {
   684→          this.bumpCooldown = RENDER3D.bumpCooldownSec;
   685→          this.vehicle.speed *= RENDER3D.bumpSlowFactor;
   686→          this.camera.addShake(RENDER3D.bumpShake);
   687→          this.audio.bump();
   688→          this.fx.spawnBurst(this.vehicle.x, this.vehicle.y, 'rgba(190,170,130,0.9)', 4, 90);
   689→          break;
   690→        }
   691→      }
   692→    }
   693→
   694→    // ---------- ترافیک (فاز 2) ----------
   695→    this.traffic.update(dt, this.vehicle);
   696→    this.resolveTrafficCollisions(dt);
   697→
   698→    // ---------- شکستنی‌ها (فاز BIG-IR) ----------
   699→    this.updateBreakables(dt);
   700→
   701→    // ---------- عابرهای پیاده (فاز BIG-IR) ----------
   702→    this.peds.update(dt, this.vehicle);
   703→
   704→    // ---------- دود ماشین آسیب‌دیده (فاز BIG-IR) ----------
   705→    if (this.vehicle.health < DAMAGE.smokeBelow) {
   706→      this.damageSmokeTimer -= dt;
   707→      if (this.damageSmokeTimer <= 0) {
   708→        const heavy = this.vehicle.health < DAMAGE.heavySmokeBelow;
   709→        this.damageSmokeTimer = heavy ? 0.09 : 0.16;
   710→        const front = this.vehicle.drawH * 0.36;
   711→        const sx = this.vehicle.x + Math.sin(this.vehicle.angle) * front;
   712→        const sy = this.vehicle.y - Math.cos(this.vehicle.angle) * front;
   713→        this.fx.spawnBurst(sx, sy, heavy ? 'rgba(60,60,64,0.8)' : 'rgba(120,120,125,0.7)', heavy ? 2 : 1, heavy ? 70 : 45);
   714→      }
   715→    }
   716→
   717→    // ---------- خاموشی موتور (فاز BIG-IR) ----------
   718→    if (this.vehicle.dead && !this.engineDeadShown) {
   719→      this.engineDeadShown = true;
   720→      this.audio.setEngineSpeed(0);
   721→      this.camera.addShake(CAM.maxShake * 0.6);
   722→      this.fx.spawnBurst(this.vehicle.x, this.vehicle.y, 'rgba(40,40,44,0.9)', 12, 120);
   723→      this.voice.trigger('breakdown', nowMs);
   724→      // اگر مأموریت فعال است، مثل تصادف سخت حساب می‌شود
   725→      this.mission.onCrash('hard');
   726→    }
   727→
   728→    // ---------- زون‌های تعمیرگاه (فاز BIG-IR) ----------
   729→    this.updateRepairZones(dt, nowMs);
   730→
   731→    // ---------- رویدادهای تصادفی (فاز 2) ----------
   732→    this.randomEvents.update(dt, this.vehicle);
   733→
   734→    // ---------- مأموریت (فاز 4) ----------
   735→    this.mission.update(dt, this.vehicle);
   736→    // دود ماشین دودکن (m02) — پف دود تا برسد تعمیرگاه
   737→    if (this.mission.getState() === 'active' && this.mission.getDef()?.id === 'm02_smoking_car') {
   738→      this.smokeTimer -= dt;
   739→      if (this.smokeTimer <= 0) {
   740→        this.smokeTimer = MISSIONS.smokeIntervalSec;
   741→        const back = this.vehicle.drawH * 0.42;
   742→        const sx = this.vehicle.x - Math.sin(this.vehicle.angle) * back;
   743→        const sy = this.vehicle.y + Math.cos(this.vehicle.angle) * back;
   744→        this.fx.spawnBurst(sx, sy, 'rgba(120,120,125,0.75)', 1, 40);
   745→      }
   746→    }
   747→
   748→    // ---------- سکه ----------
   749→    const got = this.world.collectCoins(this.vehicle.x, this.vehicle.y, 60);
   750→    if (got > 0) {
   751→      this.audio.coin();
   752→      this.fx.spawnBurst(this.vehicle.x, this.vehicle.y - 30, '#F2B705', 10, 120);
   753→      this.save.mutate((d) => { d.coins += got; });
   754→      this.voice.trigger('coin_pickup', nowMs);
   755→    }
   756→
   757→    // ---------- موتور ----------
   758→    this.audio.setEngineSpeed(Math.abs(this.vehicle.speed) / VEHICLE_PHYSICS.maxSpeed);
   759→
   760→    // ---------- رویداد پس‌زمینه ----------
   761→    this.idleTimerSec -= dt;
   762→    if (this.idleTimerSec <= 0) {
   763→      this.idleTimerSec = IDLE_EVENTS.minIntervalSec + Math.random() * (IDLE_EVENTS.maxIntervalSec - IDLE_EVENTS.minIntervalSec);
   764→      this.voice.trigger('idle_funny', nowMs);
   765→    }
   766→
   767→    // ---------- دوربین ----------
   768→    if (this.ctx) {
   769→      this.camera.halfW = RENDER.logicalWidth / 2;
   770→      this.camera.halfH = this.ctx.canvas.height / (this.scale * 2);
   771→    } else {
   772→      // حالت سه‌بعدی: نیم‌اندازه‌های منطقی صرفاً برای clamp حباب‌ها
   773→      this.camera.halfW = RENDER.logicalWidth / 2;
   774→      this.camera.halfH = 640;
   775→    }
   776→    this.camera.update(dt, this.vehicle.x, this.vehicle.y, this.vehicle.vx, this.vehicle.vy);
   777→  }

   780→  private draw(): void {
   781→    // ---------- مسیر سه‌بعدی (فاز 3D) ----------
   782→    if (this.renderer3d) {
   783→      this.draw3DFrame();
   784→      return;
   785→    }
   786→    // ---------- مسیر دوبعدی (fallback) ----------
   787→    const ctx = this.ctx;
   788→    if (!ctx || !this.renderer) return;
   789→    const cw = ctx.canvas.width;
   790→    const ch = ctx.canvas.height;
   791→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   792→    ctx.fillStyle = '#C9B98A';
   793→    ctx.fillRect(0, 0, cw, ch);
   794→
   795→    // scale منطقی: پیکسل‌های فیزیکی بر واحد منطقی
   796→    this.scale = cw / RENDER.logicalWidth;
   797→
   798→    ctx.scale(this.scale, this.scale);
   799→    this.camera.halfW = RENDER.logicalWidth / 2;
   800→    this.camera.halfH = ch / this.scale / 2;
   801→
   802→    this.renderer.draw(
   803→      ctx,
   804→      this.vehicle,
   805→      this.traffic.cars,
   806→      this.randomEvents.cat,
   807→      this.randomEvents.cones,
   808→      this.mission.getPotholes(),
   809→      this.mission.getState() === 'active' ? this.mission.getDef()?.target ?? null : null,
   810→    );
   811→    // حباب گفتار با مختصات دنیاست — باید با همان ترنسلیت دوربین رسم شود
   812→    // (renderer.draw ترنسلیت را داخل save/restore خودش دارد و بعد از restore حذف می‌شود)
   813→    ctx.save();
   814→    ctx.translate(-this.camera.offsetX(), -this.camera.offsetY());
   815→    this.drawSpeechBubble(ctx);
   816→    ctx.restore();
   817→    ctx.setTransform(1, 0, 0, 1, 0, 0);
   818→
   819→    // فلش جهت‌نمای هدف وقتی بیرون کادر است (فضای صفحه — فاز 4)
   820→    if (this.mission.getState() === 'active') {
   821→      const def = this.mission.getDef();
   822→      if (def) this.renderer.drawOffscreenArrow(ctx, def.target, this.vehicle, cw, ch, this.scale);
   823→    }
   824→  }

   869→      fps: this.fpsValue,
   870→      showFps: this.showFps,
   871→      placeholders: this.assets.getReport().placeholders.length,
   872→      mission: this.mission.hud(this.vehicle),
   873→      health: Math.round(this.vehicle.health),
   874→      engineDead: this.vehicle.dead,
   875→    });
   876→  }
   877→
   878→  // ============ رویدادهای صفحه ============
   879→  private handleResize = (): void => {
   880→    const dpr = Math.min(RENDER.maxDevicePixelRatio, window.devicePixelRatio || 1);
   881→    const w = this.canvas.clientWidth || window.innerWidth;
   882→    const h = this.canvas.clientHeight || window.innerHeight;
   883→    this.canvas.width = Math.round(w * dpr);
   884→    this.canvas.height = Math.round(h * dpr);
   885→    // رندر سه‌بعدی هم باید بداند (کانواس WebGL + اورلی + aspect دوربین)
   886→    if (this.renderer3d) this.renderer3d.resize(this.canvas.width, this.canvas.height, dpr);
   887→  };
   888→
   889→  private onKeyDown = (e: KeyboardEvent): void => {
   890→    const k = e.key.toLowerCase();
   891→    if (k === 'p' || k === 'escape') {
   892→      if (this.state === 'playing') this.pause();
   893→      else if (this.state === 'paused') this.resume();
   894→    } else if (k === 'f') {
   895→      this.toggleDebugFps();
   896→    }
   897→  };
   898→
   899→  private onVisibility = (): void => {
   900→    if (document.hidden && this.state === 'playing') this.pause();
   901→  };
   902→
   903→  private onBeforeUnload = (): void => {
   904→    this.save.flush();
   905→  };
   906→
   907→  dispose(): void {
   908→    this.disposed = true;
   909→    if (GameManager.live === this) GameManager.live = null;
   910→    cancelAnimationFrame(this.raf);
   911→    this.audio.stopEngine();
   912→    this.input.detach();
   913→    this.renderer3d?.dispose(); // حذف اورلی + آزادسازی کانتکست GPU
   914→    window.removeEventListener('resize', this.handleResize);
   915→    window.removeEventListener('keydown', this.onKeyDown);
   916→    document.removeEventListener('visibilitychange', this.onVisibility);
   917→    window.removeEventListener('beforeunload', this.onBeforeUnload);
   918→    this.save.flush();
   919→  }
   920→}
