    49→/** پنل تنظیمات صدا (در منو و Pause استفاده می‌شود) */
    50→function SettingsPanel({ game, onClose }: { game: GameInstance; onClose?: () => void }) {
    51→  const [s, setS] = useState<Settings>(game.getSettings());
    52→  const rows: Array<{ key: keyof Settings; label: string; icon: React.ReactNode }> = [
    53→    { key: "music", label: "موسیقی", icon: <Music className="h-5 w-5" /> },
    54→    { key: "sfx", label: "افکت‌ها", icon: <Volume2 className="h-5 w-5" /> },
    55→    { key: "voice", label: "جملات طنز", icon: <Mic className="h-5 w-5" /> },
    56→  ];
    57→  return (
    58→    <div className="w-full space-y-3">
    59→      {rows.map((r) => (
    60→        <div key={r.key} className="flex items-center gap-3">
    61→          <span className="text-[#3a2e2a]">{r.icon}</span>
    62→          <span className="w-24 text-sm font-bold text-[#3a2e2a]">{r.label}</span>
    63→          <input
    64→            type="range"
    65→            className="fc-range flex-1"
    66→            min={0}
    67→            max={1}
    68→            step={0.05}
    69→            value={s[r.key]}
    70→            aria-label={`صدای ${r.label}`}
    71→            onChange={(e) => {
    72→              const v = Number(e.target.value);
    73→              setS((p) => ({ ...p, [r.key]: v }));
    74→              game.setSetting(r.key, v);
    75→            }}
    76→          />
    77→          <span className="w-10 text-center text-sm font-bold text-[#3a2e2a]">{fa(Math.round(s[r.key] * 100))}٪</span>
    78→        </div>
    79→      ))}
    80→      <div className="flex items-center gap-3 opacity-50">
    81→        <span className="text-sm font-bold text-[#3a2e2a]">لرزش گوشی</span>
    82→        <span className="text-xs text-[#3a2e2a]">(فازهای بعدی)</span>
    83→      </div>
    84→      {onClose && (
    85→        <button
    86→          type="button"
    87→          className="fc-btn mt-2 w-full rounded-2xl bg-[#3a2e2a] py-2.5 text-white"
    88→          onPointerDown={(e) => {
    89→            e.preventDefault();
    90→            onClose();
    91→          }}
    92→        >
    93→          بستن
    94→        </button>
    95→      )}
    96→    </div>
    97→  );
    98→}
